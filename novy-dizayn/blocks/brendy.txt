      const el = document.getElementById("brand-" + this.slugAnchor(this.state.picked));
      if (el) {

      }
    }
    Object.keys(this.logoRefs).forEach(k => {
      const box = this.logoRefs[k].current;
      if (!box) return;
      box.querySelectorAll("img[data-url]").forEach(el => {
        const u = el.getAttribute("data-url");
        if (!u || u.indexOf("{{") >= 0) { el.style.display = "none"; return; }
        // адрес появился — снимаем скрытие с первого прохода, иначе логотип не покажется никогда
        el.style.display = "block";
        if (el.getAttribute("src") !== u) el.setAttribute("src", u);
      });
    });

  // площадь уравнивает их визуально — разброс ровно ×1.00 по всем 26 брендам.
  // Ограничители убраны 25.08.2026: они были нужны из-за почти квадратного логотипа
  // Royal Thermo (0.83:1), который при равной площади вырастал в столбик. Владелец
  // заменил его на широкий (1.71:1) — теперь весь диапазон пропорций 1.45…6.58
  // укладывается в высоту 28…60 px и ширину 86…185 px без всякой обрезки.
  // Если появится бренд с пропорцией вне этого диапазона — вернуть ограничители.
  logoBox(name) {
    const r = (this.state.ratio || {})[name];
    if (!r) return { w: 150, h: 34 };            // пропорция неизвестна — прежний размер
    // ручная поправка из data/brand-logo-scale.json — для логотипов с мелким
    // вшитым текстом, который при равной площади не читается
    const k = (this.state.logoScale || {})[name] || 1;
    const h = Math.sqrt(5200 * k / r);
    return { w: Math.round(h * r), h: Math.round(h) };
  }
