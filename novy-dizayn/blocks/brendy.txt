      const nb = (src[b].groups || []).reduce((s, g) => s + (g.n || 0), 0);
      return nb - na;
    });
    names.forEach(n => { this.logoRefs[n] = this.logoRefs[n] || React.createRef(); });

    const brandsTotal = (() => {
      const idx = this.state.idx || [];
      const set = new Set();
      idx.forEach(g => (g.brandList || []).forEach(b => set.add(b)));
      const n = set.size;
      const a = n % 10, b = n % 100;
      const w = (a === 1 && b !== 11) ? "бренд" : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? "бренда" : "брендов";
      return n ? n + " " + w : "больше двадцати брендов";
    })();

    const rows = names.map(n => ({
      name: n,
      anchor: "#brand-" + this.slugAnchor(n),
      country: STRANA[n] || "—",
      groupsText: this.groupsText(src[n].groups || []),
      href: src[n].top === "multisplit" ? "/multisplit?brand=" + encodeURIComponent(n) : "/katalog?g=" + src[n].top + "&brand=" + encodeURIComponent(n)
    }));

    // указатель «бренды по группам товаров»: из тех же brand-cards.json
    const titles = {};
    names.forEach(n => (src[n].groups || []).forEach(g => { if (!titles[g.slug]) titles[g.slug] = g.t; }));
    const order = PORYADOK.concat(Object.keys(titles).filter(s => PORYADOK.indexOf(s) < 0));
    const gidx = order.filter(s => titles[s]).map(s => ({
      title: titles[s].charAt(0).toUpperCase() + titles[s].slice(1),
      href: GRUPPY_STR[s] || "/katalog?g=" + s,
      brands: names.filter(n => (src[n].groups || []).some(g => g.slug === s)).map(n => ({ name: n, href: "#brand-" + this.slugAnchor(n) }))
    }));

    return {
      brandsTotal,
      gidx,
      rows,
      cards: names.map(n => ({
        name: n,
        anchor: "brand-" + this.slugAnchor(n),
        cardBg: n === this.state.picked ? "#F7FBFF" : "#fff",
        cardBorder: n === this.state.picked ? "#0066FF" : "#DCE5F0",
        logo: logos[n] || "",
        noLogo: !logos[n],
        hasLogo: !!logos[n],
        logoW: this.logoBox(n).w,
        logoH: this.logoBox(n).h,
        logoRef: this.logoRefs[n],
        about: ABOUT[n] || "",
        geo: GEO[n] || "",
        hasGeo: !!GEO[n],
        hasExtra: !!EXTRA[n],
        extraLabel: EXTRA[n] ? EXTRA[n][0] : "",
        extraHref: EXTRA[n] ? EXTRA[n][1] : "",
        href: src[n].top === "multisplit"
          ? "/multisplit?brand=" + encodeURIComponent(n)
          : "/katalog?g=" + src[n].top + "&brand=" + encodeURIComponent(n),
        // подпись описывает ровно то, что откроется: одна группа, а не весь бренд
        linkLabel: n + " в группе «" + ((src[n].groups || [])[0] || {}).t + "» →",
        groups: (src[n].groups || []).slice(0, 12).map(g => ({
          title: g.t.charAt(0).toUpperCase() + g.t.slice(1),
          sub: (g.st ? "в наличии" : "под заказ") + (g.note ? " · " + g.note : ""),
          subColor: g.st ? "#178841" : "#64748B",
          href: g.slug === "multisplit"
            ? "/multisplit?brand=" + encodeURIComponent(n)
            : "/katalog?g=" + g.slug + "&brand=" + encodeURIComponent(n) + (g.st ? "&stock=in" : "")
        }))
      }))
    };
  }

  // группы бренда в таблице: до пяти названий, без счётчиков
  groupsText(groups) {
    const ts = groups.map(g => g.t);
    const s = ts.slice(0, 5).join(", ");
    return s.charAt(0).toUpperCase() + s.slice(1) + (ts.length > 5 ? " и другие группы" : "");
  }
}
</script>
