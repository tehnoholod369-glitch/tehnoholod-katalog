function calcAir(){
  const room = NORMS.rooms[$('room').value];
  const S = parseFloat($('area').value) || 0;
  const H = parseFloat($('height').value) || 0;
  const N = $('people').value === '' ? null : parseInt($('people').value, 10);
  const krOwn = $('kr').value === '' ? null : parseFloat($('kr').value);
  const windows = $('tgWindows').classList.contains('on');
  const zapas = Math.max(0, parseFloat($('zapas').value) || 0);
  // Расчётный объём Vp по приложению Г СП РК 4.02-101-2012:
  // «для помещений высотой 6 м и более следует принимать Vp = 6·A».
  // Без этой оговорки склад с потолком 10 м получал бы завышенный расход.
  const Vgeom = S * H;
  const V = H >= 6 ? S * 6 : Vgeom;
  const capped = V !== Vgeom;
  const rows = [];

  // 1. по кратности — формула Г.5: L = Vp · n
  const kr = krOwn !== null ? krOwn : room.n;
  if (kr !== null && kr !== undefined) {
    rows.push({
      key: 'krat',
      name: 'По кратности (Г.5)',
      formula: 'L = Vp &times; n = ' + fmt(V) + ' м³ &times; ' + kr + ' 1/ч'
             + (capped ? ' <span style="color:#a4501a">(Vp = 6&times;S: потолок ' + H + ' м &ge; 6 м)</span>' : ''),
      value: V * kr,
      src: krOwn !== null ? 'задано вручную' : room.nSrc,
      badge: krOwn !== null ? 'orient' : (room.nBadge || 'orient')
    });
  } else {
    rows.push({key:'krat', name:'По кратности (Г.5)', formula:'кратность для этого типа не задана', value:null,
      src:'нет проверенного норматива — задайте кратность вручную', badge:'check'});
  }

  // 2. по людям
  const per = windows ? room.perPerson : room.perPersonNoWin;
  if (N !== null && N > 0 && per) {
    rows.push({
      key: 'people',
      name: 'По количеству людей (Г.7)',
      formula: 'L = N &times; m = ' + N + ' чел. &times; ' + per + ' м³/ч',
      value: N * per,
      src: room.perSrc, badge: room.perBadge || 'check'
    });
  } else {
    rows.push({key:'people', name:'По количеству людей (Г.7)', formula: N===null||N===0 ? 'людей не указано — метод не применялся' : 'норма на человека для этого типа не задана',
      value:null, src:'—', badge:null});
  }

  // 3. по площади
  if (room.perArea) {
    rows.push({
      key: 'area',
      name: 'По площади помещения (Г.6)',
      formula: 'L = A &times; k = ' + fmt(S) + ' м² &times; ' + room.perArea + ' м³/(ч·м²)',
      value: S * room.perArea,
      src: room.areaSrc, badge: room.areaBadge || 'check'
    });
  } else {
    rows.push({key:'area', name:'По площади помещения (Г.6)', formula:'норма на м² для этого типа не задана', value:null, src:'—', badge:null});
  }

  const vals = rows.filter(r => r.value !== null).map(r => r.value);
  const base = vals.length ? Math.max.apply(null, vals) : null;
  const req = base === null ? null : base * (1 + zapas / 100);
  const winner = base === null ? null : rows.find(r => r.value === base).key;
  return {room, S, H, V, Vgeom, capped, N, rows, base, req, zapas, winner};
}

  if (items.length === 1) notes.push('Все три варианта совпали — в каталоге под этот расход одна подходящая позиция.');
  if (items.length > 1 && budget && opt && budget.p < opt.p) {
    notes.push('Ближайший по расходу дороже варианта с минимальной ценой на ' + fmt(opt.p - budget.p) + ' ₸ при разнице по расходу ' + fmt(Math.abs(budget.q - opt.q)) + ' м³/ч. Ближайший типоразмер не означает «самый выгодный» — сравните оба.');
  }
  return {items, below: [], notes, relaxed, poolSize: pool.length};
}

Г.1. /* ==== обратная связь ====
   Смысл: жалоба «калькулятор посчитал ерунду» без вводных бесполезна — воспроизвести нечего.
   Кнопка собирает весь вход, весь расчёт и весь подбор в текст, который можно вставить куда угодно. */
function sendFeedback(a, res){
  const L = [];
  L.push('ПОДБОР ПВУ — сообщение о проблеме');
  L.push('Сборка калькулятора: ' + BUILD.date + ', позиций в базе: ' + BUILD.models);
  L.push('');
  L.push('ВВОД: ' + a.room.title + ' · ' + fmt(a.S) + ' м² · h ' + a.H + ' м' +
         (a.N ? ' · ' + a.N + ' чел.' : ' · людей не указано') +
         ' · окна: ' + ($('tgWindows').classList.contains('on') ? 'есть' : 'нет') +
         ' · запас ' + a.zapas + ' %' +
         ' · система: ' + $('sys').options[$('sys').selectedIndex].text +
         ' · только в наличии: ' + ($('tgStock').classList.contains('on') ? 'да' : 'нет'));
  L.push('');
  L.push('РАСЧЁТ:');
  a.rows.forEach(r => L.push('  ' + r.name + ': ' +
      r.formula.replace(/&times;/g, 'x').replace(/&larr;/g, '<-').replace(/<[^>]+>/g, '') +
      ' = ' + (r.value === null ? '—' : Math.round(r.value) + ' м³/ч')));
  L.push('  ИТОГО требуется: ' + Math.round(a.req) + ' м³/ч');
  L.push('');
  L.push('ПОДБОР:');
  if (!res.items.length) {
    L.push('  ничего не подобрано (просмотрено ' + res.poolSize + ' позиций)');
  } else {
    res.items.forEach(it => L.push('  ' + it.role + ': ' + [it.m.b, it.m.m].filter(Boolean).join(' ') +
        ' | ' + (it.m.ns || 'без НС') + ' | ' + Math.round(it.m.q) + ' м³/ч | ' + fmt(it.m.p) + ' ₸ | ' + (it.m.st || '')));
  }
  L.push('');
  L.push('ЧТО НЕ ТАК: (напишите здесь)');
  const txt = L.join('\n');
  const done = () => {
    const b = $('fbBtn');
    b.textContent = 'Скопировано ✓ — вставьте в WhatsApp';
    setTimeout(() => { b.textContent = 'Что-то не так в подборе — сообщить'; }, 4000);
  };
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(txt).then(done, () => fallback(txt, done));
  } else {
    fallback(txt, done);
  }
}
function fallback(txt, done){
  const ta = document.createElement('textarea');
  ta.value = txt; ta.style.position = 'fixed'; ta.style.opacity = '0';
  document.body.appendChild(ta); ta.select();
  try { document.execCommand('copy'); done(); }
  catch (e) { window.prompt('Скопируйте текст вручную (Ctrl+C):', txt); }
  ta.remove();
}

function buildKp(m, qty, a){
  return {
    customer: ($('customer').value || '').trim() || 'не указан',
    object: [(($('objaddr').value || '').trim() || null),
             a.room.title + ' ' + fmt(a.S) + ' м², h ' + a.H + ' м',
             'требуемый расход ' + fmt(a.req) + ' м³/ч'].filter(Boolean).join(', '),
    price_includes_vat: true,
    vat_rate: 0.16,
    equipment: [{ name: [m.b, m.m].filter(Boolean).join(' ') + (m.ns ? ' (' + m.ns + ')' : ''),
                  qty: qty, price: m.p }],
    _расчёт: a.rows.filter(r => r.value !== null)
              .map(r => r.name + ': ' + r.formula.replace(/&times;/g,'x').replace(/<[^>]+>/g,'') + ' = ' + Math.round(r.value) + ' м³/ч')
              .concat(['принят максимум + запас ' + a.zapas + ' % = ' + Math.round(a.req) + ' м³/ч',
                       'подобрано: расход установки ' + Math.round(m.q) + ' м³/ч',
                       'наличие на дату сборки ' + BUILD.date + ': ' + (m.st || 'нет данных')]),
    _оговорки: [
      'Аэродинамический расчёт сети не выполнялся — напор установки требует проверки по сети объекта.',
      'Монтаж и пусконаладка в файл не включены: цена называется после осмотра объекта.',
      'Цена — на ' + BUILD.date + '; перед оформлением уточните актуальность.'
    ],
    _комментарий: 'Сформировано калькулятором подбора ПВУ ИП «ТехноХолод», сборка ' + BUILD.date + '.'
  };
}

document.addEventListener('click', function(e){

  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = 'КП_' + String(m.m || 'позиция').replace(/[^\wА-Яа-яЁё\-]+/g, '_').slice(0, 60) + '.json';
  document.body.appendChild(a); a.click(); a.remove();
  b.textContent = 'Файл сохранён ✓';
  setTimeout(() => { b.textContent = 'Выгрузить в КП (JSON)'; }, 2500);
});
