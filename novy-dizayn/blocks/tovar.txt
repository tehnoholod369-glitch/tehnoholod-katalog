          // ON/OFF того же бренда в каталоге нашёлся только у 4 моделей из 45,
          // у остальных в «переплату» входит ещё и разница брендов (26.08.2026)
          // Вторая строка — при равной эффективности. Именно она отвечает на вопрос
          // «сколько стоит сам приток»: компрессор у обеих моделей одного класса.
          paybackNote: (briz && cheapestOnoff)
            ? "Сколько кубов в час даёт встроенный приток, изготовитель не публикует, поэтому по производительности мы приборы не равняем."
              + (sameClassInv ? "" : " Инвертора того же класса и мощности без притока в каталоге нет, поэтому вторая строка не показана.")
            : "",
          // заголовок блока называет то, с чем сравниваем: иначе связка
          // «кондиционер + приточка» стояла бы под шапкой «ближайший ON/OFF»
          paybackTitle: briz && cheapestOnoff
            ? "Сравнение со связкой «кондиционер + приточка»"
            : "Сравнение с ближайшим ON/OFF из каталога",
          // Названия эталонов — ссылки на их карточки: покупатель должен иметь
          // возможность проверить, с чем мы сравниваем, а не верить на слово.
          // Поэтому текст разложен на части: разметка соберёт его со ссылками.
          cmpFresh: !!(briz && cheapestOnoff),
          cmpPlain: !!(!briz && years > 0 && cheapestOnoff),
          cmpLead1: briz ? "Дешевле всего собрать то же самое так: " : "Сравнили с ",
          cmpName1: cheapestOnoff ? (cheapestOnoff.b + (cheapestOnoff.s ? " " + cheapestOnoff.s : "")) : "",
          cmpHref1: cheapestOnoff ? ("/tovar?g=" + s.group + "&sl=" + (cheapestOnoff.sl || "")) : "",
          cmpMid1: " (ON/OFF) плюс бризер ",
          brizName: briz ? briz.model : "",
          brizHref: briz ? ("/tovar?g=" + (briz.g || "ventilyaciya") + "&sl=" + (briz.sl || "")) : "",
          cmpTail1: briz
            ? " на " + (briz ? briz.flow : 0) + " м³/ч — вместе " + this.fmt(baseCost) + " ₸, "
              + (priceGap > 0 ? "эта модель дороже на " + this.fmt(priceGap) + " ₸"
                              : "эта модель дешевле на " + this.fmt(-priceGap) + " ₸")
              + ". Но там неинверторный компрессор, так что в разницу входит не только приток."
            : (cheapestOnoff
              ? " — это ON/OFF того же конструктива и мощности из нашего каталога, он дешевле на "
                + this.fmt(priceGap) + " ₸. При таком режиме работы разница возвращается за "
                + String(years).replace(".", ",") + " " + this.seasonWord(years) + "."
                + (cheapestOnoff.b !== it.b ? " Бренд другой, так что в разницу входит не только инвертор." : "")
              : ""),
          cmpLead2: "При равной эффективности: ",
          cmpName2: (briz && sameClassInv) ? (sameClassInv.b + (sameClassInv.s ? " " + sameClassInv.s : "")) : "",
          cmpHref2: (briz && sameClassInv) ? ("/tovar?g=" + s.group + "&sl=" + (sameClassInv.sl || "")) : "",
          cmpTail2: (briz && sameClassInv)
            ? " (инвертор, класс " + myCls + ") плюс тот же бризер — вместе " + this.fmt(baseCost2) + " ₸, "
              + (priceGap2 > 0 ? "эта модель дороже на " + this.fmt(priceGap2) + " ₸."
                               : "эта модель дешевле на " + this.fmt(-priceGap2) + " ₸.")
            : "",
          payback: (briz && cheapestOnoff) || years > 0 ? "1" : "",
          note: "Расчёт ориентировочный: сезон охлаждения принят за 90 дней, реальный расход зависит от жары, площади и настроек. Тариф поставьте свой — он у каждого дома свой."
        };
      })(),
      related,
      hasRelated: related.length > 0,
      analogs,
      hasAnalogs: analogs.length > 0,
      analogsNote: analogs.length
        ? (myVol ? "Тот же объём бака" : myComp === "inv" ? "Только инверторные" : myComp === "onoff" ? "Только ON/OFF" : "Тот же тип")
          + ", близкая мощность, другие бренды. Цены — из каталога, разница считается к текущей модели."
        : "",
      analogRef: this.analogRef,
      prev: () => this.go(s.idx - 1),
      next: () => this.go(s.idx + 1)
    };
  }
