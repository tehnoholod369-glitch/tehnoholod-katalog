<script type="text/x-dc" data-dc-script data-props="{}">
class Component extends DCLogic {
  // Расчёт мощности вынесен из «Контактов» 31.10.2026 (адрес /raschet-moshchnosti).
  // areaRaw и heightRaw — то, что набрано в поле, как есть: иначе «1» по пути к «12»
  // превращается под пальцем в «8», а «2,» при наборе «2,9» — в «2».
  state = { area: 25, areaRaw: "25", height: 270, heightRaw: "2,7",
            sun: "Восток / запад", glass: "Обычные окна", floor: "Не последний",
            people: "2", tech: "Нет", winter: "Только охлаждение" };

  fmt(n) { return String(Math.round(n)).replace(/\B(?=(\d{3})+(?!\d))/g, "\u2009"); }

  // «2,7» и «2.7» — одно и то же: на телефоне запятая и точка лежат на разных
  // раскладках, требовать одну — это ввод, который «не работает» без объяснения.
  чис(v) {
    const n = parseFloat(String(v).replace(",", ".").replace(/\s/g, ""));
    return isFinite(n) ? n : null;
  }

  зажать(n, min, max) { return Math.min(max, Math.max(min, n)); }

  // Кнопки шага правят и число, и то, что показано в поле: иначе после «−»
  // в поле осталось бы прежнее, набранное руками.
  шагПоля(поле, сырое, д, мин, макс) {
    this.setState(st => {
      const v = this.зажать(st[поле] + д, мин, макс);
      return { [поле]: v, [сырое]: String(v) };
    });
  }

  // Высота хранится в сантиметрах, шаг — 0,1 м, как в клиентском подборе
  шагВысоты(д) {
    this.setState(st => {
      const v = this.зажать(st.height + д, 200, 600);
      return { height: v, heightRaw: (v / 100).toFixed(1).replace(".", ",") };
    });
  }

  renderVals() {
    const s = this.state;

    const V = s.area * s.height / 100;                       // объём, м³
    const SUN = { "Север / тень": 30, "Восток / запад": 35, "Юг, много солнца": 40 };
    const PEOPLE = { "1": 1, "2": 2, "3": 3, "4": 4, "5 и больше": 5 };
    const TECH = { "Нет": 0, "Телевизор или ноутбук": 150, "Компьютер, несколько устройств": 450, "Кухня (холодильник, плита)": 700 };
    const qV = SUN[s.sun] || 35;
    const bigGlass = s.glass === "Большие окна, больше 2 м²";
    const glassW = bigGlass ? (s.sun === "Юг, много солнца" ? 250 : 150) : 0;
    const floorA = s.floor === "Последний этаж" ? 0.10 : 0;
    const envW = V * qV * (1 + floorA) + glassW;
    const peopleN = PEOPLE[s.people] || 2;
    const peopleW = peopleN * 100;
    const techW = TECH[s.tech] || 0;
    const total = envW + peopleW + techW;
    const heatWinter = s.winter === "Нужен обогрев зимой";
    // диапазон выбора: −5 % … +15 % к расчётной нагрузке
    const kwLow = total * 0.95, kwHigh = total * 1.15;
    const kw1 = w => (Math.round(w / 100) / 10).toFixed(1).replace(".", ",");
    const req = Math.round(total * 3.412);

    const CLASSES = [7000, 9000, 12000, 18000, 24000, 36000, 48000, 60000];
    // допуск -5 %: класс чуть ниже расчёта не гоним на ступень вверх (Royal Clima: -5 %...+15 %)
    const need = req * 0.95;
    const clsBtu = CLASSES.find(c => c >= need);
    const clsIdx = CLASSES.indexOf(clsBtu);
    const overload = !clsBtu;
    const cls = clsBtu ? { 7000: "07", 9000: "09", 12000: "12", 18000: "18", 24000: "24", 36000: "36", 48000: "48", 60000: "60" }[clsBtu] : "60+";
    const btu = clsBtu ? this.fmt(clsBtu) : "60 000+";

    // Текст расчёта собираем один раз: он уходит и в WhatsApp, и (без контактов)
    // в «События» приёмника — чтобы было видно, по каким площадям считают.
    const расчётТекст = "Расчёт по подбору. Площадь " + s.area + " м²"
      + ", потолок " + (s.height / 100).toFixed(1).replace(".", ",") + " м"
      + ", окна: " + s.sun.toLowerCase() + (glassW ? ", " + s.glass.toLowerCase() : "")
      + (floorA ? ", последний этаж" : "") + ", людей: " + s.people
      + (techW ? ", техника: " + s.tech.toLowerCase() : "") + (heatWinter ? ", нужен обогрев зимой" : "")
      + ". " + (overload ? "Нагрузка " + kw1(total) + " кВт — выше бытового ряда, нужен расчёт на замере."
                         : "Нагрузка " + kw1(kwLow) + "–" + kw1(kwHigh) + " кВт, нужен класс " + cls + " (" + btu + " BTU).");

    // Пояснение: почему именно этот класс, и что будет с соседними — вместо общей фразы
    const classKw = b => b / 3412;
    let explainText;
    if (overload) {
      explainText = "Для помещения " + s.area + " м² с выбранными условиями нагрузка " + kw1(total) + " кВт — выше бытового ряда, одним бытовым блоком её не закрыть. Нужна мультисплит- или полупромышленная система, мощность посчитаем на замере.";
    } else {
      const parts = ["Нагрузка около " + kw1(total) + " кВт, в сильную жару до " + kw1(kwHigh) + ". Класс " + cls + " даёт ≈ " + kw1(classKw(clsBtu) * 1000) + " кВт."];
      if (clsIdx > 0) {
        const lo = CLASSES[clsIdx - 1], loKw = classKw(lo) * 1000;
        const short = Math.round((1 - loKw / total) * 100);
        parts.push("Класс " + (lo / 1000 < 10 ? "0" : "") + lo / 1000 + " (≈ " + kw1(loKw) + " кВт) " + (short <= 10
          ? "почти хватает, но в самую жару будет работать на пределе."
          : "не хватит: на " + short + " % меньше нагрузки."));
      }
      if (clsIdx === 0 && classKw(clsBtu) * 1000 > total * 1.6) {
        parts.push("Это самый малый класс в ряду: запас большой, но меньшего не бывает. Инвертор при малой нагрузке сбавляет мощность.");
      } else if (clsIdx < CLASSES.length - 1) {
        const hi = CLASSES[clsIdx + 1], hiKw = classKw(hi) * 1000;
        if (hiKw > total * 1.4) {
          parts.push("Класс " + (hi / 1000 < 10 ? "0" : "") + hi / 1000 + " (≈ " + kw1(hiKw) + " кВт) — запас " + Math.round((hiKw / total - 1) * 100) + " %: дороже без выигрыша в комфорте.");
        }
      }
      if (heatWinter) {
        parts.push("Расчёт — на охлаждение. Для обогрева зимой смотрите в карточке мощность в режиме тепла и нижний температурный предел работы: на сильном морозе она заметно ниже номинальной.");
      }
      explainText = parts.join(" ");
    }
    const num2 = x => String(Math.round(x * 100) / 100).replace(".", ",");
    const breakdownRows = [
      { k: "Помещение: " + s.area + " м² × " + (s.height / 100).toFixed(1).replace(".", ",") + " м = " + this.fmt(Math.round(V)) + " м³, по " + qV + " Вт/м³", v: this.fmt(V * qV) + " Вт" },
      { k: "Последний этаж", v: floorA ? "+10 % → " + this.fmt(V * qV * (1 + floorA)) + " Вт" : "нет" },
      { k: "Большие окна, больше 2 м²", v: glassW ? "+" + glassW + " Вт" : "нет" },
      { k: "Люди: " + peopleN + " × 100 Вт", v: this.fmt(peopleW) + " Вт" },
      { k: "Техника: " + (techW ? s.tech.toLowerCase() : "нет"), v: this.fmt(techW) + " Вт" },
      { k: "Итого нагрузка", v: this.fmt(total) + " Вт" },
      { k: "Диапазон выбора (−5 % … +15 %)", v: kw1(kwLow) + "–" + kw1(kwHigh) + " кВт" },
      { k: "Класс (допуск −5 %)", v: overload ? "60 000+ BTU" : btu + " BTU" }
    ];

    const chooser = (title, key, options) => ({
      title,
      options: options.map(o => ({
        label: o,
        pick: () => this.setState({ [key]: o }),
        border: s[key] === o ? "#0066FF" : "#DCE5F0",
        bg: s[key] === o ? "#E9F4FF" : "#fff",
        fg: s[key] === o ? "#0033A0" : "#64748B"
      }))
    });

    return {
      area: s.area,
      areaRaw: s.areaRaw,
      height: s.height,
      heightRaw: s.heightRaw,
      areaMinus: () => this.шагПоля("area", "areaRaw", -1, 8, 180),
      areaPlus: () => this.шагПоля("area", "areaRaw", 1, 8, 180),
      heightMinus: () => this.шагВысоты(-10),
      heightPlus: () => this.шагВысоты(10),
      onAreaText: e => {
        const raw = e.target.value, n = this.чис(raw);
        this.setState(n === null ? { areaRaw: raw }
          : { areaRaw: raw, area: this.зажать(Math.round(n), 8, 180) });
      },
      onAreaBlur: () => this.setState(st => ({ areaRaw: String(st.area) })),
      onHeightText: e => {
        const raw = e.target.value, n = this.чис(raw);
        this.setState(n === null ? { heightRaw: raw }
          : { heightRaw: raw, height: Math.round(this.зажать(n, 2, 6) * 100) });
      },
      onHeightBlur: () => this.setState(st => ({
        heightRaw: (st.height / 100).toFixed(1).replace(".", ",")
      })),
      choices: [
        chooser("Сторона света окон", "sun", ["Север / тень", "Восток / запад", "Юг, много солнца"]),
        chooser("Остекление", "glass", ["Обычные окна", "Большие окна, больше 2 м²"]),
        chooser("Этаж", "floor", ["Не последний", "Последний этаж"]),
        chooser("Сколько людей обычно в комнате", "people", ["1", "2", "3", "4", "5 и больше"]),
        chooser("Техника в комнате", "tech", ["Нет", "Телевизор или ноутбук", "Компьютер, несколько устройств", "Кухня (холодильник, плита)"]),
        chooser("Режим работы", "winter", ["Только охлаждение", "Нужен обогрев зимой"])
      ],

      kwLabel: kw1(kwLow) + "–" + kw1(kwHigh) + " кВт",
      calcHref: "https://wa.me/77000369369?text=" + encodeURIComponent(расчётТекст),
      // Контактов здесь не спрашиваем, значит это не заявка, а событие: пишем
      // площадь, потолок, поправки и класс — по ним видно, с каким запросом
      // приходят. Переход в WhatsApp не задерживаем, ответа не ждём.
      onCalcSend: () => {
        if (!window.TH_LEAD) { return; }
        window.TH_LEAD.event({
          channel: "подбор на /raschet-moshchnosti", ev: "расчёт отправлен в WhatsApp",
          val: overload ? "выше бытового ряда" : cls + " · " + btu + " BTU",
          area: s.area, height: (s.height / 100).toFixed(1).replace(".", ","),
          corr: [s.sun, glassW ? s.glass : "", floorA ? s.floor : "", "людей " + s.people,
                 techW ? s.tech : "", heatWinter ? s.winter : ""].filter(Boolean).join(", ")
        });
      },

      classLabel: overload ? "60 000+ BTU" : cls + " · " + btu + " BTU",
      modelsLabel: overload ? "Показать полупромышленные системы" : "Показать модели класса " + cls,
      // кнопка обещает класс — значит и ведёт в выдачу ровно этого класса: каталог
      // с 27.08.2026 фильтрует по классу мощности и понимает ?btu=. Через площадь
      // было неточно — полоса «20–35 м²» одинаково отвечала и на 09, и на 12.
      // Нагрузку выше бытового ряда бытовая выдача не закрывает — там сразу полупром.
      // Класс 36 и выше — это 10 кВт холода и больше: в бытовой выдаче таких моделей
      // четыре, в полупромышленной — 81. Ведём туда, где товар есть, иначе кнопка
      // «показать модели класса 36» открыла бы бытовую выдачу целиком.
      modelsHref: overload
        ? "/poluprom"
        : "/katalog?g=" + (clsBtu >= 36000 ? "poluprom" : "bytovye") + "&btu=" + clsBtu,
      explain: explainText,
      breakdown: breakdownRows
    };
  }
}
