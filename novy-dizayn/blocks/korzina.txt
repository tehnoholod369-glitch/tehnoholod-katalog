    list.forEach(x => { if (!this.boxRefs[x.sl]) this.boxRefs[x.sl] = React.createRef(); });

      if (!box) return;
      box.querySelectorAll("img[data-url]").forEach(el => {
        const u = el.getAttribute("data-url");
        if (!u || u.indexOf("{{") >= 0 || el.getAttribute("src") === u) return;
        // одна попытка со вторым хранилищем, прежде чем прятать пустой прямоугольник
        el.onerror = () => {
          const alt = window.photoAlt ? window.photoAlt(el.getAttribute("src")) : "";
          if (alt && el.getAttribute("src") !== alt) { el.onerror = null; el.setAttribute("src", alt); }
          else el.style.display = "none";
        };
        el.setAttribute("src", u);
      });
    });
  }

  componentDidUpdate() { this.syncPhotos(); }

  ph(u) { return u ? (window.photoUrl ? window.photoUrl(u) : u) : ""; }

  renderVals() {
    const s = this.state;
    const items = s.items || [];
    const cart = window.CART;

    const rows = items.map(x => {
      const qty = x.q || 1;
      const inStock = /налич/i.test(x.st || "");
      return {
        brand: x.b, sku: x.s ? "арт. " + x.s : "артикул уточняется", name: x.n,
        spec: (s.titles || {})[x.g] || "",
        stock: x.st || "уточняется",
        stockFg: inStock ? "#0E9F6E" : "#B25200",
        stockBg: inStock ? "#E7F7F0" : "#FFF1E3",
        img: this.ph(x.i), boxRef: this.boxRefs[x.sl], qty,
        href: "/tovar?g=" + encodeURIComponent(x.g || "") + "&sl=" + encodeURIComponent(x.sl || ""),
        sum: x.p ? this.fmt(x.p * qty) + " ₸" : "цена по запросу",
        each: !x.p ? "посчитаем при подтверждении" : qty > 1 ? "по " + this.fmt(x.p) + " ₸" : "с НДС",
        inc: () => cart && cart.setQty(x.sl, qty + 1),
        dec: () => cart && cart.setQty(x.sl, Math.max(1, qty - 1)),
        remove: () => cart && cart.remove(x.sl)
      };
    });

    const equip = items.reduce((n, x) => n + (x.p || 0) * (x.q || 1), 0);
    const totalQty = items.reduce((n, x) => n + (x.q || 1), 0);
    const byRequest = items.filter(x => !x.p).length;

    const svc = (key, title, text, price) => {
      const on = s[key];
      return {
        title, text, price, on,
        toggle: () => this.setState({ [key]: !on }),
        border: on ? "#B9DBF7" : "#DCE5F0",
        bg: on ? "#E6F2FD" : "#fff",
        boxBorder: on ? "#0872D3" : "#DCE5F0",
        boxBg: on ? "#0872D3" : "#fff"
      };
    };

    const chip = (key, label) => ({
      label,
      pick: () => this.setState({ [key]: label }),
      border: s[key] === label ? "#0872D3" : "#DCE5F0",
      bg: s[key] === label ? "#E6F2FD" : "#fff",
      fg: s[key] === label ? "#0A4EA7" : "#5A6675"
    });

    const deliveryNotes = {
      "Алматы, до адреса": "Доставка по городу — стоимость уточняется при подтверждении заказа, зависит от адреса и габаритов.",
      "Другой город Казахстана": "Отправка транспортной компанией по её тарифу. Срок называем при подтверждении — зависит от города и наличия позиции.",
      "Самовывоз": "Забрать можно после подтверждения наличия. Часть позиций идёт под заказ и приходит в Алматы в срок поставки."
    };
    const paymentNotes = {
      "Kaspi или перевод": "Оплата на счёт в АО «Kaspi Bank» или переводом по реквизитам. Отгрузка после поступления оплаты.",
      "Рассрочка 12 месяцев": "Оформляет банк: Kaspi, ForteBank, Halyk. Решение по одобрению — на стороне банка. Монтаж в рассрочку не входит.",
      "Счёт для юрлица": "Выставим счёт с НДС, договор и закрывающие документы. Понадобятся наименование и ИИН/БИН."
    };

    const isCompany = s.payment === "Счёт для юрлица";
    const installment = s.payment === "Рассрочка 12 месяцев";

    const form = {
      name: s.name.trim(), phone: s.phone.trim(), city: s.city.trim(),
      address: s.address.trim(), company: isCompany ? s.company.trim() : "",
      bin: isCompany ? s.bin.trim() : "", comment: s.comment.trim(),
      mount: s.mount, kit: s.kit, delivery: s.delivery, payment: s.payment
    };
    const phoneOk = form.phone.replace(/[^\d]/g, "").length >= 10;
    const ready = items.length > 0 && form.name.length > 1 && phoneOk;

    return {
      items: rows,
      isEmpty: items.length === 0,
      notEmpty: items.length > 0,
      clear: () => { if (cart) cart.clear(); },
      hasChanges: (s.changes || []).length > 0,
      changes: s.changes || [],

      services: [
        svc("mount", "Монтаж под ключ", "Установка блоков, вакуумирование, пусконаладка, уборка. Срок 1–3 дня после доставки.", "после замера"),
        svc("kit", "Соединительный комплект и трасса", "Медные трубы, изоляция, кабель, дренаж. Длина зависит от расположения наружного блока.", "по длине трассы")
      ],
      deliveryOptions: Object.keys(deliveryNotes).map(k => chip("delivery", k)),
      deliveryNote: deliveryNotes[s.delivery],
      paymentOptions: Object.keys(paymentNotes).map(k => chip("payment", k)),
      paymentNote: paymentNotes[s.payment],
      isCompany,

      fName: s.name, fPhone: s.phone, fCity: s.city, fAddress: s.address,
      fCompany: s.company, fBin: s.bin, fComment: s.comment,
      onName: e => this.setState({ name: e.target.value }),
      onPhone: e => this.setState({ phone: e.target.value }),
      onCity: e => this.setState({ city: e.target.value }),
      onAddress: e => this.setState({ address: e.target.value }),
      onCompany: e => this.setState({ company: e.target.value }),
      onBin: e => this.setState({ bin: e.target.value }),
      onComment: e => this.setState({ comment: e.target.value }),
      nameBorder: !form.name || form.name.length > 1 ? "#DCE5F0" : "#E11A13",
      phoneBorder: !form.phone || phoneOk ? "#DCE5F0" : "#E11A13",
      formNote: ready
        ? "Всё заполнено — заказ уйдёт в WhatsApp составом, останется нажать «Отправить»."
        : "Имя и телефон нужны, чтобы перезвонить и подтвердить наличие. Остальное — по желанию.",

      totalQty,
      equipTotal: equip ? this.fmt(equip) + " ₸" : "0 ₸",
      // ставка НДС 16 %: сумма уже с налогом, выделяем его обратным счётом
      vat: equip ? this.fmt(equip * 16 / 116) + " ₸" : "0 ₸",
      hasByRequest: byRequest > 0,
      byRequestLine: byRequest + " шт. — посчитаем",
      kitLine: s.kit ? "по длине трассы" : "не нужен",
      mountLine: s.mount ? "после замера" : "не нужен",
      deliveryLine: s.delivery === "Самовывоз" ? "бесплатно" : "уточняется",
      showInstallment: installment && equip > 0,
      // рассрочка = розница x 1,175, делится на 12 месяцев без округления до 990
      monthly: this.fmt(equip * 1.175 / 12) + " ₸/мес",

      submitLabel: !items.length ? "Добавьте позиции из каталога"
        : !ready ? "Заполните имя и телефон"
        : isCompany ? "Запросить счёт" : "Отправить заказ в WhatsApp",
      orderHref: ready && cart ? cart.orderHref(form) : "",
      orderTarget: ready ? "_blank" : "",
      orderClick: e => { if (!ready) e.preventDefault(); },
      orderBg: ready ? "#0872D3" : "#8FB6E4",
      orderCursor: ready ? "pointer" : "not-allowed",
      preview: cart && items.length ? cart.orderText(form) : ""
    };
  }
}
