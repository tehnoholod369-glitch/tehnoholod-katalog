        : isCompany ? "Запросить счёт" : "Отправить заказ в WhatsApp",
      // href оставлен запасным путём: если JS-обработчик не сработает, ссылка
      // всё равно откроет WhatsApp с составом — просто без номера.
      orderHref: ready && cart ? cart.orderHref(form) : "",
      orderTarget: ready ? "_blank" : "",
      // Заказ сперва уходит в приёмник и получает номер, потом открывается WhatsApp.
      // Окно открываем ДО ожидания ответа: открытое после await блокируется
      // как попап. Приёмник не ответил — уходим без номера, продажу не теряем.
      orderClick: e => {
        if (!ready || s.sending) { e.preventDefault(); return; }
        e.preventDefault();
        const w = window.open("", "_blank");
        this.setState({ sending: true });
        const done = (no) => {
          this.setState({ sending: false, orderNo: no || "" });
          const href = cart.orderHref(form, no);
          if (w) { w.location.href = href; } else { window.location.href = href; }
        };
        if (cart && cart.submitOrder) { cart.submitOrder(form).then(done).catch(() => done("")); }
        else { done(""); }
      },
      orderBg: ready ? "#0872D3" : "#8FB6E4",
      orderCursor: ready ? "pointer" : "not-allowed",
      preview: cart && items.length ? cart.orderText(form) : ""
    };
  }
}
</script>
