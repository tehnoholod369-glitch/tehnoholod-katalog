/* ---------- сквозной номер КП ----------
   Номер выдаёт приёмник заявок (лист «КП», счётчик КП-ГГММ-NNNN), а не браузер:
   у менеджера с телефона и с компьютера браузерные счётчики разошлись бы, а после
   чистки кэша начались бы заново. Решение владельца 04.09.2026.

   Спрашиваем номер в момент выгрузки, а не при открытии: иначе каждый заход
   сжигал бы номер. Один раз выданный номер остаётся у этого КП — повторная
   выгрузка второй номер не берёт.

   Адрес приёмника берём из cart.js (CART.exec): второго адреса не заводим,
   при смене развёртывания менять в одном месте. */
var номерВыдан = false;
function выдатьНомер(){
  if(номерВыдан || !ПОЗ.length) return Promise.resolve(true);
  var exec = (window.CART && window.CART.exec) || "";
  if(!exec) return Promise.resolve(false);
  var сум = всего(), ндс = сум - Math.round(сум/(1+НДС));
  var тело = { kind:"kp", customer:$("kpКлиент").value.trim() || "До востребования",
    bin:$("kpБин").value.trim(), phone:$("kpТел").value.trim(),
    until:$("kpДо").value, vat:ндс, src:"/kp",
    ua:(navigator.userAgent||"").slice(0,120),
    items: ПОЗ.map(function(p){ return {art:p.арт, name:p.имя, qty:+p.кол||0,
      price:ценаПозиции(p)}; }) };
  return fetch(exec, { method:"POST", headers:{"Content-Type":"text/plain;charset=utf-8"},
                       body: JSON.stringify(тело) })
    .then(function(r){ return r.json(); })
    .then(function(d){
      if(d && d.number){ $("kpНомер").value = d.number; номерВыдан = true; бланк(); сохранить(); return true; }
      return false;
    })
    .catch(function(){ return false; });
}
function сНомером(дальше){
  выдатьНомер().then(function(ок){
    // Молча ставить свой номер нельзя: два КП с одинаковым номером хуже, чем
    // пауза и честное предупреждение.
    if(!ок && !$("kpНомер").value.trim()){
      if(!confirm("Не удалось получить номер КП с сайта (нет связи с приёмником).\n"
                + "Выгрузить без номера? Номер можно вписать вручную.")) return;
    }
    дальше();
  });
}

function текст(){
  var сум = всего(), безНДС = Math.round(сум/(1+НДС)), ндс = сум - безНДС;
  var L = ["Коммерческое предложение № "+$("kpНомер").value+" от "+датаСловами(new Date()), ОРГ.имя, ""];
  ПОЗ.forEach(function(p,i){
    L.push((i+1)+". "+p.имя);
    L.push("   "+p.кол+" "+p.ед+" x "+деньги(ценаПозиции(p))+" = "+деньги(суммаПозиции(p)));
    if(p.ссылка) L.push("   "+p.ссылка);
    L.push("");
  });
  L.push("Итого с НДС: "+деньги(сум)+" тг (в т.ч. НДС "+(НДС*100)+" % — "+деньги(ндс)+" тг)");
  L.push(прописью(сум)); L.push("");
  условия().forEach(function(s){ L.push(s); });
  L.push(""); L.push("Каталог и цены: "+ОРГ.сайт);
  L.push(ОРГ.тел+"  ·  "+ОРГ.почта);
  return L.join("\n");
}

function копироватьТекст(){
  var t = текст();
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(t).then(function(){ alert("Текст КП скопирован."); });
  } else {
    var e = document.createElement("textarea"); e.value = t; document.body.appendChild(e);
    e.select(); document.execCommand("copy"); e.parentNode.removeChild(e); alert("Текст КП скопирован.");
  }
}
