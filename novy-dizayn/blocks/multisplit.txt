        current: k.brand === brand,
        border: k.brand === brand ? "#0872D3" : "#DCE5F0",
        bg: k.brand === brand ? "#E6F2FD" : "#fff",
        pick: () => this.setState({ brand: k.brand, rooms: s.rooms.map(r => ({ area: r.area, type: r.type })) })

      hasBrandRows: brandRows.length > 1,
      brandRows,
      brandsRef: this.brandsRef,
      brandCompareNote: priced.length
        ? "Одна и та же конфигурация — " + this.plural(rooms.length, "комната", "комнаты", "комнат")
          + ", те же площади и типы блоков — собрана в " + this.plural(priced.length, "бренде", "брендах", "брендах")
          + " из " + brandList.length + "." + (missNote ? " Не собрались: " + missNote + "." : "")
        : "",
      savingLine: (currentKit && cheapest && currentKit.total > cheapest.total)
        ? "У " + cheapest.brand + " та же конфигурация дешевле на " + this.money(currentKit.total - cheapest.total) + "."
        : (currentKit && cheapest && currentKit.brand === cheapest.brand)
          ? "Выбранный бренд — самый доступный из собравшихся."
          : "",
      galRef: this.galRef,
      outRef: this.outRef,
      specHref: "https://wa.me/77000369369?text=" + encodeURIComponent(
        "Заявка на мультисплит, бренд " + brand + ". "
        + rooms.map(r => r.title + ": " + r.area + " м², " + r.typeLabel + (r.model ? ", " + r.model : "")).join("; ")
        + ". Режим: " + s.mode + "."),
      modalRef: this.modalRef,
      hasModal: !!md,
      closeModal: () => this.setState({ modal: null }),
      mdImg: md ? (window.photoUrl ? window.photoUrl(md.i) : md.i) : "",
      mdTitle: md ? md.b + " " + md.s : "",
      mdName: md ? md.n : "",
      mdNoPhoto: !!(md && !md.i),
      mdPhotoNote: md && md.i ? "1 фото из 3 — второй ракурс и пульт пришлём по запросу" : "",
      mdManual: md && md.man ? md.man : "",
      mdRows: md ? [
        { k: "Тип блока", v: md.t || "нет данных", c: md.t ? "#18212F" : "#B25200" },
        { k: "Мощность", v: md.btu ? md.btu + " BTU" : "нет данных", c: md.btu ? "#18212F" : "#B25200" },

        // 26.08.2026: строка была зашита текстом и данные не читала вовсе, хотя

        { k: "Габариты, вес", v: [md.dim, md.wt].filter(Boolean).join(" · ") || "уточним при расчёте",

      overLimit: blocked,
      simulCaveat: allPriced && s.mode === "По очереди" && out && out.area && totalArea > out.area
        ? "Модуль рассчитан на " + out.area + " м², а комнат суммарно " + totalArea + " м². В режиме «по очереди» этого достаточно; если все комнаты будут работать одновременно в жару — мощности не хватит, переключите режим и пересчитайте."
        : "",
      overTitle: altBrand
        ? "В линейке " + brand + " такого модуля нет"
        : planBuilt
          ? (splitPlan.count > 1 ? "Одним наружным модулем эту конфигурацию не закрыть" : "В линейке " + brand + " закрываются не все комнаты")
          : "Мультисплит такую конфигурацию не закроет",
      overText: (!altBrand && planBuilt)
        ? "Нужен модуль от " + needOut + " BTU на " + this.plural(ports, "порт", "порта", "портов")
          + ", а максимум в линейке " + brand + " — " + brandMaxBtu + " BTU. Вариант ниже: " + planLabel + "."
        : altBrand
          ? "Нужен модуль от " + needOut + " BTU на " + this.plural(ports, "порт", "порта", "портов")
            + ", а в линейке " + brand + " максимум " + brandMaxBtu + " BTU. Такой комплект собирается у " + altBrand + " — переключите бренд выше."
          : "Нужен модуль от " + needOut + " BTU на " + this.plural(ports, "порт", "порта", "портов")
            + ", а самый мощный наружный модуль мультисплита в каталоге — " + anyMaxBtu + " BTU (до " + anyMaxArea + " м²). Ограничение не в числе блоков, а в модуле: больше мощности мультисплит не даёт. Для " + totalArea + " м² нужна полупромышленная или мультизональная система.",
      overCta: (planBuilt || altBrand) ? "" : "Полупромышленное оборудование →",
      overBg: (planBuilt || altBrand) ? "#F5FAFF" : "#FDECEA",
      overBorder: (planBuilt || altBrand) ? "#B9DBF7" : "#F8CBBE",
      overFg: (planBuilt || altBrand) ? "#0A4EA7" : "#C21610",
      hasSplit: planBuilt,
      splitTitle: splitPlan
        ? (splitPlan.count === 1
            ? "Вариант: мультисплит на " + this.plural(rooms.length - splitPlan.singles, "комнату", "комнаты", "комнат")
              + " + " + this.plural(splitPlan.singles, "отдельная система", "отдельные системы", "отдельных систем")
            : "Вариант: " + this.plural(splitPlan.count, "мультисплит-система", "мультисплит-системы", "мультисплит-систем")
              + (splitPlan.singles ? " + " + this.plural(splitPlan.singles, "отдельная система", "отдельные системы", "отдельных систем") : ""))
        : "",
      weakWarn: weakRooms.length
        ? "Разбиение на несколько систем не предлагаем: " + this.plural(weakRooms.length, "комнате", "комнатам", "комнатам")
          + " не хватает мощности самого крупного блока линейки. Такую площадь делят на два блока в одной комнате или закрывают полупромышленной системой."
        : "",
      splitLines: splitPlan ? splitPlan.lines : [],
      splitSubtitle: planBuilt
        ? (splitPlan.count > 1
            ? "Комнаты делятся на группы, каждая группа получает свой наружный модуль. Блоки внутри остаются те же."
            : "Одна мультисплит-система на " + this.plural(rooms.length - splitPlan.singles, "комнату", "комнаты", "комнат")
              + ", остальные закрываем отдельными кондиционерами по каталогу.")
        : "",
      splitTotal: planBuilt ? (splitPlan.total || splitPlan.partial) : "",
      splitNote: splitPlan ? splitPlan.note : "",
      sliderNote: "Ползунок до 80 м² — это максимум, который закрывает один внутренний блок мультисплита (24 000 BTU). Комнату больше делят на два блока или ставят полупром.",
      roomCount: this.plural(cmpRooms.length, "комнате", "комнатам", "комнатам")
        + (skipIdx.length ? " из " + rooms.length : ""),
      outCardDisplay: planBuilt ? "none" : "flex",
      sepFrom: sepFull ? "от " + this.money(sepMin) : "считаем",
      sepNote: sepFull
        ? "Самые доступные модели каталога тех же классов, оборудование без монтажа."
        : "Подберём по вашим комнатам.",
      verdict,
      spec: rooms.map(r => {
        const moved = planBuilt && skipIdx.indexOf(r.idx) >= 0;
        return {
          k: r.title + " · " + r.typeLabel,
          v: moved
            ? r.area + " м² → в мультисплит-часть не входит, считаем отдельной системой"
            : r.area + " м² → " + (r.model || "подбор") + (r.price ? " · " + this.money(r.price) : ""),
          color: moved || !r.price ? "#B25200" : "#18212F"
        };
      }).concat(
        planBuilt
          ? splitPlan.specRows
          : [{ k: "Наружный модуль", v: !out ? "подбор" : outOk ? out.s + " · " + out.btu + " BTU" + (out.p ? " · " + this.money(out.p) : "") : "не подобран в линейке " + brand, color: outOk && out && out.p ? "#18212F" : "#B25200" }]
      ).concat([
        // наличие комплекта — по худшему звену: модуль под заказ означает ожидание поставки
        (() => {
          const parts = rooms.map(r => r.stockOk).filter(x => x !== undefined);
          const blocksOk = parts.length && parts.every(Boolean);
          const bothOk = outOk && out && out.ok && blocksOk;
          const blocksTxt = blocksOk ? "блоки в наличии" : "часть блоков под заказ";
          const outTxt = !out ? "модуль не подобран"
            : !outOk ? "модуля нужной мощности в линейке нет"
            : out.ok ? "модуль в наличии" : "наружный модуль под заказ";
          return {
            k: "Наличие комплекта",
            v: bothOk ? "блоки и модуль в наличии" : outTxt + ", " + blocksTxt,
            color: bothOk ? "#0E9F6E" : "#B25200"
          };
        })(),
        { k: "Режим использования", v: s.mode, color: "#18212F" },
        {
          k: "Оборудование итого",
          v: allPriced ? this.money(total)
            : planOk ? splitPlan.total
            : planBuilt ? splitPlan.partial + " + отдельный кондиционер"
            : (outOk ? "по запросу" : "комплект не собирается"),
          color: (allPriced || planOk) ? "#18212F" : "#B25200"
        },
        { k: "Трассы, комплект, монтаж", v: "по факту замера", color: "#B25200" }
      ])
    };
  }
