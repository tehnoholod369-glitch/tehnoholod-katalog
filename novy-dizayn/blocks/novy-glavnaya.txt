  g(slug) {
    return ((this.state && this.state.idx) || []).find(x => x.slug === slug) || null;
  }

  cnt(slug) {
    const g = this.g(slug);
    return g ? this.fmt(g.count) : "—";
  }

  plural(n, one, few, many) {
    const a = n % 10, b = n % 100;
    if (a === 1 && b !== 11) return n + " " + one;
    if (a >= 2 && a <= 4 && (b < 10 || b >= 20)) return n + " " + few;
    return n + " " + many;
  }

  // Разряды в числах. На Главной они стояли только у цен (одна строка), а счётчики
  // печатались голыми: в герое было «3347 позиций в каталоге». На остальных страницах
  // редизайна разрядка есть везде — приводим к общему виду. Тонкий пробел, как в каталоге.
  fmt(n) { return String(Math.round(n)).replace(/\B(?=(\d{3})+(?!\d))/g, " "); }

  heroStats() {
    const idx = (this.state && this.state.idx) || [];
    const total = idx.reduce((n, g) => n + g.count, 0);
    const stock = idx.reduce((n, g) => n + g.inStock, 0);
    const brands = new Set();
    idx.forEach(g => (g.brandList || []).forEach(b => brands.add(b)));
    const tail = (n, one, few, many) => this.plural(n, one, few, many).replace(String(n) + " ", "");
    return total
      ? [
          { v: this.fmt(total), k: tail(total, "позиция", "позиции", "позиций") + " в каталоге" },
          { v: this.fmt(idx.length), k: tail(idx.length, "группа", "группы", "групп") + " техники" },
          { v: this.fmt(brands.size), k: tail(brands.size, "бренд", "бренда", "брендов") }
        ]
      : [{ v: "—", k: "загружаем каталог" }, { v: "—", k: "" }, { v: "—", k: "" }];
  }

  mi(label, slug) {
    const g = this.g(slug);
    return {
      label,
      count: g ? this.fmt(g.count) : "",
      href: "/katalog?g=" + slug,
      subs: g && (g.subs || []).length > 1
        ? g.subs.filter(s => s.name.toLowerCase() !== label.toLowerCase() && this.своя(slug, s)).slice(0, 5).map(s => ({
            label: s.name,
            n: s.n,
            href: "/katalog?g=" + slug + "&sub=" + encodeURIComponent(s.name)
          }))
        : []
    };
  }
