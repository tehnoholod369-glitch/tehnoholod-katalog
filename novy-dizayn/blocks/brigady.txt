В рейтинг входят только оценки, подтверждённые «Сервис-картой 369». Оценку компании смотрите в Google и 2ГИС, ссылки на странице «Отзывы».</div>

<script type="text/x-dc" data-dc-script data-props="{}">
class Component extends DCLogic {
  PRIEMNIK = "https://script.google.com/macros/s/AKfycbzqoF8JVLc-OjrSLJbwZ8oBi0MbC5p89VHmLY9di3rcaK0TrTdEEPAduBqbYPRqWPwdgA/exec";

  state = { data: null, failed: false, dir: "", city: "", open: {}, form: {}, sent: {} };

  componentDidMount() {
    const q = new URLSearchParams(window.location.search);
    this.setState({ dir: q.get("napravlenie") || "", city: q.get("gorod") || "" });

    try {
      const cached = sessionStorage.getItem("th_brigady");
      if (cached) { this.setState({ data: JSON.parse(cached) }); }
    } catch (_) {}

    fetch(this.PRIEMNIK + "?data=brigady", { cache: "no-cache" })
      .then(r => r.ok ? r.json() : null)
      .then(d => {
        if (!d || !d.brigady) { this.setState({ failed: true }); return; }
        this.setState({ data: d, failed: false });
        // Пустой ответ в кэш не кладём: иначе один пустой ответ заставит страницу
        // весь визит утверждать «бригад нет» — даже когда реестр уже заполнен.
        if (d.brigady.length) {
          try { sessionStorage.setItem("th_brigady", JSON.stringify(d)); } catch (_) {}
        }
      })
      // Молча пустую страницу не показываем: пустота читается как «бригад нет»,
      // а это неправда — это сбой связи.
      .catch(() => this.setState({ failed: true }));
  }

  componentDidUpdate() {
    // Карточки появляются после загрузки, поэтому раскрываем нужную не по таймеру,
    // а когда она реально отрисована.
    const hash = decodeURIComponent((window.location.hash || "").slice(1));
    if (hash && this.state.data && !this.jumped) {
      const el = document.getElementById(hash);
      if (el) {
        this.jumped = true;
        this.setState({ open: Object.assign({}, this.state.open, { [hash]: true }) });
        window.scrollTo({ top: window.pageYOffset + el.getBoundingClientRect().top - 16, behavior: "smooth" });
      }
    }
  }

  /** Фильтр держим и в адресе: ссылку на «вентиляционные бригады» должно быть
   *  можно послать клиенту, а не только нажать самому. */
  setFilter(patch) {
    this.setState(patch);
    try {
      const s = Object.assign({ dir: this.state.dir, city: this.state.city }, patch);
      const q = new URLSearchParams();
      if (s.dir) { q.set("napravlenie", s.dir); }
      if (s.city) { q.set("gorod", s.city); }
      const qs = q.toString();
      window.history.replaceState(null, "", window.location.pathname + (qs ? "?" + qs : ""));
    } catch (_) {}
  }

  votesLabel(n) {
    const d100 = n % 100, d10 = n % 10;
    if (d100 >= 11 && d100 <= 14) { return n + " оценок"; }
    if (d10 === 1) { return n + " оценка"; }
    if (d10 >= 2 && d10 <= 4) { return n + " оценки"; }
    return n + " оценок";
  }

  crewsWord(n) {
    const d100 = n % 100, d10 = n % 10;
    if (d100 >= 11 && d100 <= 14) { return n + " бригад"; }
    if (d10 === 1) { return n + " бригада"; }
    if (d10 >= 2 && d10 <= 4) { return n + " бригады"; }
    return n + " бригад";
  }

  /** «5» вместо «5,0» рядом с «4,8» читается как другая шкала. Всегда один знак. */
  ratingLabel(x) { return x.toFixed(1).replace(".", ","); }

  dateLabel(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso || ""));
    return m ? m[3] + "." + m[2] + "." + m[1] : String(iso || "");
  }

  chip(on) {
    return on
      ? { bg: "#0872D3", fg: "#fff", border: "#0872D3", countFg: "#BFDCF7" }
      : { bg: "#fff", fg: "#2C3846", border: "#DCE5F0", countFg: "#8593A6" };
  }

  form(cert) { return this.state.form[cert] || { score: 0, name: "", card: "", text: "" }; }

  setForm(cert, patch) {
    this.setState({ form: Object.assign({}, this.state.form,
      { [cert]: Object.assign({}, this.form(cert), patch) }) });
  }

  /** Отправка отзыва. Рапорт об отправке — только рядом с настоящей отправкой:
   *  sendBeacon, а если он не сработал — fetch. Гейт витрины проверяет это. */
  sendReview(cert) {
    const f = this.form(cert);
    if (!(f.score >= 1 && f.score <= 5) || f.text.trim().length < 2) { return; }
    const body = new URLSearchParams({
      kind: "review", brigada: cert, score: String(f.score),
      name: f.name.trim(), text: f.text.trim(), karta: f.card.trim(),
      src: (document.referrer || "").slice(0, 180) || "direct",
      ses: (() => { try { let s = sessionStorage.getItem("th_ses");
        if (!s) { s = Math.random().toString(36).slice(2, 10); sessionStorage.setItem("th_ses", s); }
        return s; } catch (_) { return ""; } })(),
      ua: (navigator.userAgent || "").slice(0, 260)
    }).toString();
    const ct = "application/x-www-form-urlencoded;charset=UTF-8";
    let ok = false;
    if (navigator.sendBeacon) {
      try { ok = navigator.sendBeacon(this.PRIEMNIK, new Blob([body], { type: ct })); } catch (_) { ok = false; }
    }
    if (!ok) {
      fetch(this.PRIEMNIK, { method: "POST", headers: { "Content-Type": ct },
                             body: body, mode: "no-cors", keepalive: true });
    }
    this.setState({ sent: Object.assign({}, this.state.sent, { [cert]: true }) });
  }

  renderVals() {
    const d = this.state.data;
    const all = (d && d.brigady) || [];
    const loading = !d && !this.state.failed;
    const dir = this.state.dir, city = this.state.city;
    // Порог публикации рейтинга живёт в приёмнике (PUB_MIN) — здесь только подпись.
    const pubMin = (d && d.pubMin) || 5;

    const list = all.filter(c =>
      (!dir || (c.dirs || []).indexOf(dir) >= 0) &&
      (!city || c.region === city));

    // Фишки строим только по тому, что реально есть в реестре: направление,
    // по которому нет ни одной бригады, — обещание фильтра без содержимого.
    const dirSeen = [];
    all.forEach(c => (c.dirs || []).forEach((k, i) => {
      const label = (c.dirLabels || [])[i];
      if (!label) { return; }
      const hit = dirSeen.find(x => x.key === k);
      if (hit) { hit.n++; } else { dirSeen.push({ key: k, label: label, n: 1 }); }
    }));
    const citySeen = [];
    all.forEach(c => {
      const hit = citySeen.find(x => x.key === c.region);
      if (hit) { hit.n++; } else { citySeen.push({ key: c.region, label: c.city, n: 1 }); }
    });
    citySeen.sort((a, b) => b.n - a.n);

    return {
      isLoading: loading,
      // «Бригад нет» — утверждение о реальности, и опирается оно только на успешный
      // ответ. Если ответа не было, честная формулировка — «не загрузилось».
      isError: this.state.failed && all.length === 0,
      isEmpty: !this.state.failed && !!d && all.length === 0,
      hasAny: all.length > 0,
      nothingFound: all.length > 0 && list.length === 0,
      manyCities: citySeen.length > 1,
      pubMinLabel: this.votesLabel(pubMin).replace(/^\d+\s*/, pubMin + " "),
      crewsTitle: list.length ? this.crewsWord(list.length) + (dir || city ? " по фильтру" : " в реестре")
                              : "Бригады",
      updatedNote: d && d.updated ? "реестр на " + this.dateLabel(d.updated) : "",
      resetFilters: () => this.setFilter({ dir: "", city: "" }),

      dirChips: [{ key: "", label: "Все", n: all.length }].concat(dirSeen).map(x => Object.assign({
        label: x.label,
        count: x.n,
        pick: () => this.setFilter({ dir: x.key })
      }, this.chip(dir === x.key))),

      cityChips: [{ key: "", label: "Все города", n: all.length }].concat(citySeen).map(x => Object.assign({
        label: x.label,
        count: x.n,
        pick: () => this.setFilter({ city: x.key })
      }, this.chip(city === x.key))),

      crews: list.map(c => {
        const open = !!this.state.open[c.cert];
        const sent = !!this.state.sent[c.cert];
        const f = this.form(c.cert);
        const ready = f.score >= 1 && f.text.trim().length >= 2;
        const facts = [
          c.city     ? { k: "ВЫЕЗД", v: c.city + (c.zone ? " " + c.zone : "") } : null,
          c.warranty ? { k: "ГАРАНТИЯ НА МОНТАЖ", v: c.warranty + " мес — даёт бригада" } : null,
          c.since    ? { k: "В СЕТИ", v: "с " + c.since + " года" } : null
        ].filter(Boolean);

        return {
          cert: c.cert,
          dirLine: (c.dirLabels || []).join(" · ") || "Направление уточняется",
          cardBorder: open ? "#B9DBF7" : "#DCE5F0",
          hasRating: c.rating !== null,
          accruing: !!c.accruing,
          accruingNote: "уже " + this.votesLabel(c.votes) + " из " + pubMin,
          noRating: c.rating === null && !c.accruing,
          rating: c.rating === null ? "" : this.ratingLabel(c.rating),
          votesLabel: this.votesLabel(c.votes),
          facts: facts,

          // Пожелание, а не бронь: заявка уходит нам, бригаду назначаем мы —
          // с учётом пожелания. Телефона бригады на странице нет.
          callHref: "https://wa.me/77000369369?text=" + encodeURIComponent(
            "Здравствуйте! На монтаж хочу бригаду " + c.cert + "."),

          open: open,
          toggle: () => this.setState({ open: Object.assign({}, this.state.open, { [c.cert]: !open }) }),
          toggleLabel: open ? "Свернуть" : (c.reviewsTotal > 0 ? "Отзывы" : "Оставить отзыв"),

          reviews: (c.reviews || []).map(r => ({
            name: r.name, date: this.dateLabel(r.date), text: r.text,
            verified: !!r.v, plain: !r.v,
            stars: "★".repeat(r.score) + "☆".repeat(5 - r.score)
          })),
          noReviews: (c.reviews || []).length === 0,

          formSent: sent,
          formOpen: !sent,
          formName: f.name,
          formCard: f.card,
          formText: f.text,
          scoreButtons: [1, 2, 3, 4, 5].map(n => ({
            label: "★ " + n,
            pick: () => this.setForm(c.cert, { score: n }),
            bg:     f.score === n ? "#0872D3" : "#fff",
            fg:     f.score === n ? "#fff"    : "#2C3846",
            border: f.score === n ? "#0872D3" : "#DCE5F0"
          })),
          onName: e => this.setForm(c.cert, { name: e.target.value }),
          onCard: e => this.setForm(c.cert, { card: e.target.value }),
          onText: e => this.setForm(c.cert, { text: e.target.value }),
          send: () => this.sendReview(c.cert),
          sendBg: ready ? "#0872D3" : "#8FB6E4",
          sendCursor: ready ? "pointer" : "not-allowed",
          sendLabel: ready ? "Отправить на модерацию"
                   : (f.score < 1 ? "Поставьте оценку" : "Напишите пару слов")
        };
      })
    };
  }
}
</script>
