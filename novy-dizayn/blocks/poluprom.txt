  renderVals() {
    const s = this.state;
    const items = s.items || [];
    // Считаем по РАЗДЕЛУ выдачи (поле sub), а не по вхождению слова в тип и название:
    // «колонн» ловило и «колонный/напольный», и слово «колонный», случайно попавшее
    // в заголовок консольной модели. Раздел — то же поле, по которому фильтрует выдача,
    // поэтому число на плитке и число в выдаче теперь не могут разойтись (28.08.2026).
    const has = t => items.filter(x => (x.sub || "") === t).length;
    // «94 моделей» вместо «94 модели» — склонение считаем, а не приписываем (26.08.2026)
    const cnt = n => {
      if (!items.length) return "загружаем";
      const a = n % 10, b = n % 100;
      const w = (a === 1 && b !== 11) ? "модель"
        : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? "модели" : "моделей";
      return n + " " + w;
    };

    const kinds = Array.from(new Set(items.map(x => x.sub))).filter(Boolean).length;
    const brands = Array.from(new Set(items.map(x => x.b))).filter(Boolean).length;
    // !x.acc — то же правило, что на Главной: цена «от» считается по ОБОРУДОВАНИЮ.
    // Сейчас аксессуаров с ценой в полупроме нет и число не меняется, но условие
    // ставится заранее: заведут один расходник — и «от» уедет на его цену, как это
    // случилось на Главной с фильтрами TCL FreshIN (9 000 ₸ вместо 323 942 ₸).
    const priced = items.filter(x => x.p && !x.acc);
    const minPrice = priced.length ? Math.min.apply(null, priced.map(x => x.p)) : 0;

    const href = "/katalog?g=poluprom";
    // Плитки вели на общий адрес без фильтра: нажимаешь «Кассетные» — открывается
    // весь полупром вперемешку. Выдача принимает раздел параметром sub (28.08.2026).
    const разделHref = имя => href + "&sub=" + encodeURIComponent(имя);

    const objects = ["Магазин или торговый зал", "Офис, опенспейс", "Кафе или ресторан", "Серверная", "Цех или склад"].map(o => ({
      label: o,
      pick: () => this.setState({ object: o }),
      border: s.object === o ? "#0066FF" : "#DCE5F0",
      bg: s.object === o ? "#E9F4FF" : "#fff",
      fg: s.object === o ? "#0033A0" : "#64748B"
    }));

    const load = {
      "Магазин или торговый зал": { w: 100, note: "Витрины, освещение и поток людей дают заметный теплоприток — берём 100 Вт на м² и распределяем несколько блоков по залу, чтобы не дуло в кассира и покупателей." },
      "Офис, опенспейс": { w: 90, note: "Компьютеры и люди: 90 Вт на м². Важнее не мощность, а распределение потока — иначе часть сотрудников мёрзнет, а часть жалуется на духоту." },
      "Кафе или ресторан": { w: 120, note: "Кухня, гости и оборудование: 120 Вт на м². Зал и кухню считаем раздельно — на кухне обычная система долго не живёт." },
      "Серверная": { w: 400, note: "Оборудование работает круглосуточно: 400 Вт на м² и выше по фактической нагрузке стоек. Нужна круглогодичная работа на холод и резервирование." },
      "Цех или склад": { w: 80, note: "Высокие потолки и большой объём: 80 Вт на м², но решает не площадь, а объём и тепловыделение оборудования. Часто выгоднее канальная схема или несколько колонных." }
    };

    const conf = load[s.object];
    const kw = Math.ceil(s.area * conf.w / 100) / 10;
    const scheme = s.area <= 60 ? "1 блок" : s.area <= 150 ? "2–3 блока" : s.area <= 300 ? "3–5 блоков или канальная система" : "мультизональная система (VRF)";

    return {
      ctaLabel: items.length ? "Все " + items.length + " моделей" : "Открыть каталог",
      stats: [
        { k: "Позиций в группе", v: items.length || "—" },
        { k: "Конструктивов", v: kinds || "—" },
        { k: "Брендов", v: brands || "—" },
        { k: "Цены от", v: minPrice ? String(minPrice).replace(/\B(?=(\d{3})+(?!\d))/g, "\u2009") + " ₸" : "по запросу" }
      ],
      types: [
        {
          title: "Кассетные", count: cnt(has("Кассетные")),
          text: "Встраиваются в подвесной потолок, раздают воздух в четыре стороны. Виден только декоративный экран.",
          where: "Магазины, офисы, кафе с потолком «армстронг»",
          limit: "Нужен подвесной потолок и 25–35 см запаса над ним",
          href: разделHref("Кассетные")
        },
        {
          title: "Канальные", count: cnt(has("Канальные")),
          text: "Полностью скрыты за потолком, воздух подаётся через решётки. Можно организовать приток свежего воздуха.",
          where: "Квартиры с дизайнерским ремонтом, офисы, рестораны",
          limit: "Требуется проект разводки воздуховодов",
          href: разделHref("Канальные")
        },
        {
          title: "Напольно-потолочные", count: cnt(has("Напольно-потолочные")),
          text: "Ставятся под потолок или у пола, поток идёт вдоль поверхности. Не требуют подвесного потолка.",
          where: "Помещения без запаса по высоте, вытянутые залы",
          limit: "Блок виден в интерьере",
          href: разделHref("Напольно-потолочные")
        },
        {
          title: "Колонные", count: cnt(has("Колонные")),
          text: "Мощные напольные блоки на большой объём, ставятся у стены. Быстро охлаждают зал.",
          where: "Торговые залы, холлы, производственные помещения",
          limit: "Занимают место в помещении, шумнее кассетных",
          href: разделHref("Колонные")
        },
        {
          // Пятый конструктив добавлен 26.08.2026: три позиции считались в «Позициях
          // в группе», но карточки не имели, и сумма по карточкам не сходилась с итогом.
          // Текст по факту данных: AERONIK ASI-COHMZ, 9000–18000 BTU, все под заказ.
          title: "Консольные", count: cnt(has("Консольные")),
          text: "Компактные блоки под потолок или к полу на небольшую площадь — переходное решение между бытовым сплитом и напольно-потолочным. В каталоге AERONIK на 9000–18 000 BTU.",
          where: "Небольшие залы, кабинеты, подсобные помещения",
          limit: "Позиции идут под заказ — уточняйте срок поставки",
          href: разделHref("Консольные")
        }
      ],
      objects,
      area: s.area,
      areaRaw: s.areaRaw,
      areaMinus: () => this.шагПлощади(-10),
      areaPlus: () => this.шагПлощади(10),
      onAreaText: e => {
        const raw = e.target.value;
        const n = parseFloat(String(raw).replace(",", ".").replace(/\s/g, ""));
        this.setState(isFinite(n)
          ? { areaRaw: raw, area: Math.min(600, Math.max(30, Math.round(n))) }
          : { areaRaw: raw });
      },
      onAreaBlur: () => this.setState(st => ({ areaRaw: String(st.area) })),
      objectNote: conf.note,
      kwLabel: this.fmt(kw) + " кВт",
      scheme,
      explain: "Для объекта «" + s.object.toLowerCase() + "» площадью " + s.area + " м² при нагрузке " + conf.w + " Вт на м² нужно около " + this.fmt(kw) + " кВт холода. Точная мощность зависит от остекления, высоты потолков и оборудования — считаем на выезде.",
      steps: [
        { n: "1", title: "Выезд и обмеры", text: "Смотрим объект, замеряем, уточняем электрику и место под трассы. Для тендера — по спецификации без выезда." },
        { n: "2", title: "Схема и спецификация", text: "Считаем мощность и количество блоков, даём спецификацию с ценами и сроками поставки." },
        { n: "3", title: "Поставка и монтаж", text: "Отгружаем партиями под график работ, монтируем сертифицированными бригадами, при необходимости ночными смены." },
        { n: "4", title: "Сдача и сервис", text: "Пусконаладка, инструктаж персонала, «Сервис-карта 369» и договор на обслуживание." }
      ]
    };
  }
}
</script>
