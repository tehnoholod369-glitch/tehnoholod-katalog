class Component extends DCLogic {
  state = { bytovye: null, vent: null, updated: "" };

  componentDidMount() {
    fetch("https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/bytovye.json", {cache:"no-cache"}).then(r => r.ok ? r.json() : []).then(bytovye => this.setState({ bytovye })).catch(() => {});
    fetch("https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/ventilyaciya.json", {cache:"no-cache"}).then(r => r.ok ? r.json() : []).then(vent => this.setState({ vent })).catch(() => {});
    fetch("https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/index.json", {cache:"no-cache"}).then(r => r.ok ? r.json() : null).then(i => this.setState({ updated: (i && i.updated) || "" })).catch(() => {});
  }

  fmt(n) { return String(Math.round(n)).replace(/\B(?=(\d{3})+(?!\d))/g, " "); }

  spec(x, rx) {
    const row = (x.sp || []).find(v => rx.test(String(v[0])));
    return row ? String(row[1] || "").trim() : "";
  }

  plural(n, one, few, many) {
    const a = n % 10, b = n % 100;
    if (a === 1 && b !== 11) return n + " " + one;
    if (a >= 2 && a <= 4 && (b < 10 || b >= 20)) return n + " " + few;
    return n + " " + many;
  }

  renderVals() {
    // Числа считаются из тех же данных, что и каталог: текст не устареет от того,
    // что завтра приедет ещё одна приточная модель или уедет последняя (26.08.2026)
    const бытовые = (this.state.bytovye || []).filter(x => !x.acc);
    const вент = (this.state.vent || []).filter(x => !x.acc);

    const приточные = бытовые.filter(x => /^да$/i.test(this.spec(x, /приток/i)));
    const цены = приточные.map(x => x.p).filter(Boolean);
    const вНаличии = приточные.filter(x => /налич/i.test(x.st || "")).length;
    const бренды = Array.from(new Set(приточные.map(x => x.b).filter(Boolean))).sort();

    const установки = вент.filter(x => x.sub === "Приточные установки");
    const ценыУст = установки.map(x => x.p).filter(Boolean);

    // расход берём только там, где он в паспорте: у компактных моделей до 600 м³/ч
    const расход = установки.map(x => {
      const v = this.spec(x, /производительн|расход возд/i);
      const m = String(v).replace(/\s/g, "").match(/\d+/);
      return m ? +m[0] : 0;
    }).filter(v => v > 0 && v <= 600);

    const есть = приточные.length > 0;
    const естьУст = установки.length > 0;

    return {
      updated: this.state.updated || "выгрузку каталога",

      freshLine: есть
        ? "В каталоге " + this.plural(приточные.length, "модель", "модели", "моделей")
          + (цены.length ? " · от " + this.fmt(Math.min.apply(null, цены)) + " ₸" : "")
          + (вНаличии ? " · в наличии " + вНаличии : "")
        : "Сейчас таких моделей в каталоге нет",
      freshShort: есть
        ? this.plural(приточные.length, "модель", "модели", "моделей")
          + (бренды.length ? ": " + бренды.join(", ") : "")
        : "нет позиций",

      ventLine: естьУст
        ? "В каталоге " + this.plural(установки.length, "позиция", "позиции", "позиций")
          + (ценыУст.length ? " · от " + this.fmt(Math.min.apply(null, ценыУст)) + " ₸" : "")
        : "Сейчас таких позиций в каталоге нет",
      ventShort: естьУст ? this.plural(установки.length, "позиция", "позиции", "позиций") + " в разделе вентиляции" : "нет позиций",
      ventAir: расход.length
        ? "есть: " + Math.min.apply(null, расход) + "–" + Math.max.apply(null, расход) + " м³/ч у компактных моделей"
        : "есть, значение указано в карточке модели",

      dataNote: "Числа в тексте считаются из каталога при каждом открытии страницы — по той же выгрузке, "
        + "что и цены. Расход притока у сплит-систем не приводим: производители его не публикуют. "
        + "Стоимость монтажа называем после замера: она зависит от стены, этажа и длины трассы."
    };
  }
}
