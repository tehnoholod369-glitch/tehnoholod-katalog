<h2>Когда мульти-сплит выгоднее</h2>
<p>Мульти-сплит берут, когда на фасаде нет места под несколько наружных блоков
или их запрещает управляющая компания: один модуль обслуживает 2–5 комнат.
Внутренние блоки можно смешивать — настенные, кассетные, канальные.
Ограничение: при поломке наружного модуля перестают работать все комнаты сразу,
а суммарная мощность внутренних блоков ограничена моделью наружного.
Собрать комплект — в <a href="https://tehnoholod369.kz/multisplit">конструкторе мульти-сплита</a>.</p>

  var srcs=["https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/multisplit.json"];
  Promise.all(srcs.map(function(u){
    return fetch(u,{cache:"no-cache"}).then(function(r){return r.json()}).catch(function(){return []});
  })).then(function(all){
    var by={},min=null;
    all.forEach(function(d){(d||[]).forEach(function(x){
      if(x.s)by[x.s]=x;
      if(x.st==="В наличии"&&x.p&&(min===null||x.p<min))min=x.p;
    })});
    document.querySelectorAll(".thgeo-p").forEach(function(td){
      var x=by[td.getAttribute("data-art")];
      if(x&&x.pt)td.textContent=x.pt;
    });
    // «Цены от …» в абзаце — цена КОНКРЕТНОЙ модели, самой доступной из тех,
    // что подходят под фильтр страницы. Минимум по всей группе сюда брать нельзя:
    // на «кондиционер на 25 м²» он подставил бы модель на 7 000 BTU, которая
    // эти 25 м² не закрывает, — страница обещала бы не то (31.08.2026).
    document.querySelectorAll(".thgeo-pmin").forEach(function(el){
      var a=el.getAttribute("data-min-art"), x=a?by[a]:null, v=(x&&x.p)?x.p:null;
      if(v===null){
        [].forEach.call(document.querySelectorAll(".thgeo-p"),function(td){
          var y=by[td.getAttribute("data-art")];
          if(y&&y.p&&(v===null||y.p<v))v=y.p;
        });
      }
      if(v)el.textContent=v.toLocaleString("ru-RU").replace(/ /g," ");
    });
  });
})();
</script>
