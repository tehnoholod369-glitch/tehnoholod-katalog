  componentDidMount() {
    // Язык: ?lang=kz из ссылки, иначе выбор, сохранённый на этом устройстве.
    let язык = "ru";
    try {
      const p = new URLSearchParams(location.search);
      язык = (p.get("lang") || localStorage.getItem("th_pv_lang") || "ru") === "kz" ? "kz" : "ru";
    } catch (e) {}
    if (язык !== "ru") this.setState({ lang: язык });

    // Номер приходит из QR на бланке: ?cert=369-BR-02-001. Тогда проверяем сразу,
    // не заставляя человека набирать то, что он уже поднёс камерой.
    let из_url = "";
    try {
      const p = new URLSearchParams(location.search);
      из_url = p.get("cert") || p.get("check") || "";
    } catch (e) {}
    this.загрузитьБланки();
    if (из_url) {
      this.setState({ cert: this.norm(из_url) });
      this.check(this.norm(из_url));
    }
  }

  /** Список выложенных бланков. Отдельный запрос, а не поле приёмника:
   *  приёмник знает реестр, а что физически лежит на хранилище — знает выкат.
   *  Не ответил — кнопки просто нет, страница от этого не ломается. */
  загрузитьБланки() {
    fetch("https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/blanki.json", { cache: "no-cache" })
      .then(r => r.ok ? r.json() : null)
      .then(b => { if (b && b.items) this.setState({ blanki: b.items }); })
      .catch(() => {});
  }

  setLang(язык) {
    this.setState({ lang: язык });
    try { localStorage.setItem("th_pv_lang", язык); } catch (e) {}
  }

  /** Номер как его вводит человек. Правило одно с приёмником (нормНомер):
   *  пробел вместо дефиса и длинное тире из автозамены не должны ломать поиск. */
  norm(v) {
    return String(v || "").toUpperCase()
      .replace(/[‐-―−]/g, "-")
      .replace(/\s+/g, "-")
      .replace(/-+/g, "-")
      .replace(/^-|-$/g, "");
  }

  onCert(e) {
    this.setState({ cert: e && e.target ? e.target.value : "" });
  }

  onCheck() {
    const n = this.norm(this.state.cert);
    if (!n) return;
    this.setState({ cert: n });
    this.check(n);
  }

  check(номер) {
    this.setState({ busy: true, err: "", res: null });
    // Предел ожидания (02.09.2026). Реестр отвечает через Google Apps Script,
    // у него бывает холодный старт; без предела страница показывала
    // «Проверяем…» бесконечно, и человек с бланком в руках не получал НИКАКОГО
    // ответа. Замер 02.09.2026: обычный ответ 7 секунд, но у Apps Script
    // бывают выбросы (в замере один ответ занял 112 с). Тридцать секунд
    // покрывают нормальную работу и не заставляют ждать бесконечно.
    const прервать = ("AbortController" in window) ? new AbortController() : null;
    const срок = setTimeout(() => { if (прервать) { прервать.abort(); } }, 30000);
    const опции = { cache: "no-cache" };
    if (прервать) { опции.signal = прервать.signal; }
    fetch(this.PRIEMNIK + "?check=" + encodeURIComponent(номер), опции)
      .then(r => r.ok ? r.json() : null)
      .then(d => {
        clearTimeout(срок);
        this.setState({ res: d, busy: false, err: d ? "" : "нет ответа" });
      })
      // Связи нет — говорим об этом прямо. Молчащая кнопка читается как
      // «сертификат недействителен», а это разные вещи.
      .catch(() => {
        clearTimeout(срок);
        this.setState({ busy: false, err: "нет ответа" });
      });
  }

  /** «1 оценка · 2 оценки · 5 оценок» — число рядом с рейтингом читается
   *  человеком, а не машиной. */
  otzyvLabel(n) {
    n = Number(n) || 0;
    const a = n % 10, b = n % 100;
    const сл = (a === 1 && b !== 11) ? "оценка"
      : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? "оценки" : "оценок";
    return n + " " + сл;
  }

  /** «1 лист · 2 листа · 5 листов». Правило одно с otzyvLabel: число рядом
   *  с существительным читает человек. В казахском словаре все три формы
   *  совпадают — там число существительное не склоняет. */
  skl(n, формы) {
    n = Number(n) || 0;
    const a = n % 10, b = n % 100;
    const i = (a === 1 && b !== 11) ? 0
      : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? 1 : 2;
    return n + " " + формы[i];
  }

  dateRu(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso || ""));
    return m ? (m[3] + "." + m[2] + "." + m[1]) : (iso || "");
  }

  УРЦВЕТ_(key) { return УРЦВЕТ[key] || УРЦВЕТ.uchastnik; }

      // Виды работ по направлениям — то, что бригада заявила в анкете
      // и что мы проверили при одобрении: «Кондиционирование — монтаж».
      add(t.vidy, (d.vidy || []).map(v => v.dir + " — " + v.vidy).join("; "));
      add(t.city, d.city);
      add(t.zone, d.zone);
      add(t.dirs, (d.dirLabels || []).map(dir).join(", "));
      add(t.qual, d.qual);
      add(t.eb, d.eb);
      add(t.height, d.height);
      add(t.warranty, "12" + t.months);
      add(t.since, d.since);
      add(t.issued, this.dateRu(d.issued));
      add(t.blank, d.issue ? (t.issueWord + d.issue + (d.issueDate ? t.fromWord + this.dateRu(d.issueDate) : "")) : "");
      // Сертификаты производителей приёмник отдаёт только после того, как мы
      // видели документы. Поэтому «предъявлены нам» — факт, а не пересказ.
      if (d.certs && d.certs.list) add(t.mfr, d.certs.list + t.shown + this.dateRu(d.certs.checked));
      add(t.does, (d.skills || []).map(s => s.title).join(", "));
      if (!d.valid) hint = d.status || "";
    } else if (d && !d.ok) {
      plash = t.notFound;
      hint = (d.reason ? d.reason + ". " : "") + t.notFoundHint;
    }

    return {
      t: t,
      cert: this.state.cert,
      busy: !!this.state.busy,
      btnText: this.state.busy ? t.btnBusy : t.btn,
      hasResult: !!(plash),
      shownCert: (d && d.cert) ? d.cert : this.norm(this.state.cert),
      plash: plash, plashBg: plashBg, plashFg: plashFg,
      rows: rows, hasRows: rows.length > 0,
      trusted: trusted,
      // Разделы каталога под работы бригады — приходят из приёмника готовыми
      // (адрес + подпись), витрина их только рисует.
      // Уровень программы. Показываем только у ДЕЙСТВУЮЩЕГО сертификата:
      // «ПАРТНЁР» рядом со снятым номером читался бы как рекомендация.
      // Оценки и объём работ — только у действующего сертификата.
      hasRating: !!(d && d.ok && d.valid),
      hasRatingValue: !!(d && d.rating && d.rating.value !== null && d.rating.value !== undefined),
      ratingValue: (d && d.rating && d.rating.value !== null && d.rating.value !== undefined)
        ? String(d.rating.value).replace(".", ",") : "",
      ratingVotes: (d && d.rating) ? this.otzyvLabel(d.rating.votes) : "",
      ratingAccruing: !!(d && d.rating && d.rating.accruing),
      ratingNone: !!(d && d.rating && !d.rating.votes),
      hasWorks: !!(d && d.works && d.works.vsego),
      noWorks: !!(d && d.valid && d.works && !d.works.vsego),
      worksLine: (() => {
        const w = (d && d.works) || {};
        const tt = this.L[this.state.lang] || this.L.ru;
        const части = [];
        if (w.montaj) части.push(w.montaj + " " + tt.worksMont);
        if (w.servis) части.push(w.servis + " " + tt.worksServ);
        if (w.remont) части.push(w.remont + " " + tt.worksRem);
        return части.join(" · ");
      })(),
      // Модели: ссылка ведёт в каталог по группе и артикулу — тем же адресом,
      // что и карточка товара на витрине.
      hasTovary: !!(d && d.valid && d.tovary && d.tovary.length),
      tovary: ((d && d.tovary) || []).map(x => ({
        n: x.name,
        u: "/novy-katalog?g=" + encodeURIComponent(x.group || "")
           + "&q=" + encodeURIComponent(x.art || x.name)
      })),
      hasLevel: !!(d && d.ok && d.valid && d.level && d.level.name),
      levels: УРСТУПЕНИ.map(function (с) {
        var свой = !!(d && d.level && d.level.key === с.key);
        var цв = свой ? (УРЦВЕТ[с.key] || УРЦВЕТ.uchastnik) : УРСЕРЫЙ;
        return {
          n: с.n,
          pub: с.pub,
          bg: цв.bg, bd: цв.bd, fg: цв.fg,
          sub: свой ? "#1F2937" : "#9AA6B6"
        };
      }),
      razdely: (d && d.ok && d.valid && d.razdely) ? d.razdely : [],
      hasRazdely: !!(d && d.ok && d.valid && d.razdely && d.razdely.length),
      // Бланк показываем только действующего сертификата: у отозванного
      // бумага утверждала бы то, что страница только что опровергла.
      hasBlank: !!бланк,
      blankUrl: бланк ? бланк.u : "",
      blankLine: бланк ? (this.skl(бланк.p, t.blankPages) + " · " + бланк.kb + " КБ") : "",
      hint: hint, hasHint: !!hint,
      kzColor: this.state.lang === "kz" ? "#0066FF" : "#606F85",
      ruColor: this.state.lang === "kz" ? "#606F85" : "#0066FF",
      setKz: () => this.setLang("kz"),
      setRu: () => this.setLang("ru"),
      onCert: (e) => this.onCert(e),
      onCheck: () => this.onCheck()
    };
  }
}
</script>
