      язык = (p.get("lang") || localStorage.getItem("th_pv_lang") || "ru") === "kz" ? "kz" : "ru";
    } catch (e) {}
    if (язык !== "ru") this.setState({ lang: язык });

    // Номер приходит из QR на бланке: ?cert=369-BR-02-001. Тогда проверяем сразу,
    // не заставляя человека набирать то, что он уже поднёс камерой.
    let из_url = "";
    try {
      const p = new URLSearchParams(location.search);
      из_url = p.get("cert") || p.get("check") || "";
    } catch (e) {}
    if (из_url) {
      this.setState({ cert: this.norm(из_url) });
      this.check(this.norm(из_url));
    }
  }

  setLang(язык) {
    this.setState({ lang: язык });
    try { localStorage.setItem("th_pv_lang", язык); } catch (e) {}
  }

  /** Номер как его вводит человек. Правило одно с приёмником (нормНомер):
   *  пробел вместо дефиса и длинное тире из автозамены не должны ломать поиск. */
  norm(v) {
    return String(v || "").toUpperCase()
      .replace(/[‐-―−]/g, "-")
      .replace(/\s+/g, "-")
      .replace(/-+/g, "-")
      .replace(/^-|-$/g, "");
  }

  onCert(e) {
    this.setState({ cert: e && e.target ? e.target.value : "" });
  }

  onCheck() {
    const n = this.norm(this.state.cert);
    if (!n) return;
    this.setState({ cert: n });
    this.check(n);
  }

  check(номер) {
    this.setState({ busy: true, err: "", res: null });
    // Предел ожидания (02.09.2026). Реестр отвечает через Google Apps Script,
    // у него бывает холодный старт; без предела страница показывала
    // «Проверяем…» бесконечно, и человек с бланком в руках не получал НИКАКОГО
    // ответа. Пятнадцать секунд — заведомо больше обычных двух, но всё ещё
    // столько, сколько люди готовы ждать.
    const прервать = ("AbortController" in window) ? new AbortController() : null;
    const срок = setTimeout(() => { if (прервать) { прервать.abort(); } }, 15000);
    const опции = { cache: "no-cache" };
    if (прервать) { опции.signal = прервать.signal; }
    fetch(this.PRIEMNIK + "?check=" + encodeURIComponent(номер), опции)
      .then(r => r.ok ? r.json() : null)
      .then(d => {
        clearTimeout(срок);
        this.setState({ res: d, busy: false, err: d ? "" : "нет ответа" });
      })
      // Связи нет — говорим об этом прямо. Молчащая кнопка читается как
      // «сертификат недействителен», а это разные вещи.
      .catch(() => {
        clearTimeout(срок);
        this.setState({ busy: false, err: "нет ответа" });
      });
  }

  dateRu(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso || ""));
    return m ? (m[3] + "." + m[2] + "." + m[1]) : (iso || "");
  }

  // Цвета уровня — те же, что в карточке бригады на /brigady: клиент видит
  // одну и ту же плашку в двух местах и не гадает, одно ли это и то же.
  УРЦВЕТ_(key) { return УРЦВЕТ[key] || УРЦВЕТ.uchastnik; }

  renderVals() {
    const d = this.state.res;
    const t = this.L[this.state.lang] || this.L.ru;
    const rows = [];
    let plash = "", plashBg = "#EEF3F9", plashFg = "#5A6675", hint = "", trusted = false;

    if (this.state.err) {
      plash = t.failed;
      hint = t.failedHint;
    } else if (d && d.ok) {
      trusted = !!(d.valid && d.trusted);
      if (d.valid) { plash = t.valid; plashBg = "#E8F6EC"; plashFg = "#1B6B34"; }
      else { plash = t.invalid; plashBg = "#FDECEA"; plashFg = "#A3251B"; }

      const add = (k, v) => { if (v) rows.push({ k: k, v: String(v) }); };
      const dir = (x) => t.dirMap[x] || x;
      add(t.who, d.type === "мастер" ? t.master : t.brigada);
      if (d.type === "мастер") { add(t.fio, d.fio); add(t.inBrigada, d.brigada); }
      add(t.chief, d.chief);
      add(t.city, d.city);
      add(t.zone, d.zone);
      add(t.dirs, (d.dirLabels || []).map(dir).join(", "));
      add(t.qual, d.qual);
      add(t.eb, d.eb);
      add(t.height, d.height);
      add(t.warranty, d.warranty ? (d.warranty + t.months) : "");
      add(t.since, d.since);
      add(t.issued, this.dateRu(d.issued));
      add(t.checked, d.checked ? (this.dateRu(d.checked) + (d.checkedDays != null ? " · " + d.checkedDays + t.daysAgo : "")) : "");
      add(t.blank, d.issue ? (t.issueWord + d.issue + (d.issueDate ? t.fromWord + this.dateRu(d.issueDate) : "")) : "");
      // Сертификаты производителей приёмник отдаёт только после того, как мы
      // видели документы. Поэтому «предъявлены нам» — факт, а не пересказ.
      if (d.certs && d.certs.list) add(t.mfr, d.certs.list + t.shown + this.dateRu(d.certs.checked));
      add(t.does, (d.skills || []).map(s => s.title).join(", "));
      if (!d.valid) hint = d.status || "";
    } else if (d && !d.ok) {
      plash = t.notFound;
      hint = (d.reason ? d.reason + ". " : "") + t.notFoundHint;
    }

    return {
      t: t,
      cert: this.state.cert,
      busy: !!this.state.busy,
      btnText: this.state.busy ? t.btnBusy : t.btn,
      hasResult: !!(plash),
      shownCert: (d && d.cert) ? d.cert : this.norm(this.state.cert),
      plash: plash, plashBg: plashBg, plashFg: plashFg,
      rows: rows, hasRows: rows.length > 0,
      trusted: trusted,
      // Разделы каталога под работы бригады — приходят из приёмника готовыми
      // (адрес + подпись), витрина их только рисует.
      // Уровень программы. Показываем только у ДЕЙСТВУЮЩЕГО сертификата:
      // «Партнёр 369» рядом со снятым номером читался бы как рекомендация.
      hasLevel: !!(d && d.ok && d.valid && d.level && d.level.name),
      levelName: (d && d.level && d.level.name) || "",
      levelPub: (d && d.level && d.level.pub) || "",
      hasLevelSince: !!(d && d.valid && d.level && d.level.since),
      levelSince: (d && d.level && d.level.since)
        ? ((this.L[this.state.lang] || this.L.ru).levelSinceWord || "") + this.dateRu(d.level.since)
        : "",
      levelBg: УРЦВЕТ[(d && d.level && d.level.key) || "uchastnik"].bg,
      levelBd: УРЦВЕТ[(d && d.level && d.level.key) || "uchastnik"].bd,
      levelFg: УРЦВЕТ[(d && d.level && d.level.key) || "uchastnik"].fg,
      razdely: (d && d.ok && d.valid && d.razdely) ? d.razdely : [],
      hasRazdely: !!(d && d.ok && d.valid && d.razdely && d.razdely.length),
      hint: hint, hasHint: !!hint,
      kzColor: this.state.lang === "kz" ? "#0872D3" : "#8593A6",
      ruColor: this.state.lang === "kz" ? "#8593A6" : "#0872D3",
      setKz: () => this.setLang("kz"),
      setRu: () => this.setLang("ru"),
      onCert: (e) => this.onCert(e),
      onCheck: () => this.onCheck()
    };
  }
}
</script>
