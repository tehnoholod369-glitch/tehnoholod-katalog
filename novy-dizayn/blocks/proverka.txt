class Component extends DCLogic {
  // Приёмник заявок — тот же, что у страницы бригад. Второго адреса не заводим.
  PRIEMNIK = "https://script.google.com/macros/s/AKfycbzqoF8JVLc-OjrSLJbwZ8oBi0MbC5p89VHmLY9di3rcaK0TrTdEEPAduBqbYPRqWPwdgA/exec";

  state = { cert: "", busy: false, res: null, err: "" };

  componentDidMount() {
    // Номер приходит из QR на бланке: ?cert=369-BR-02-001. Тогда проверяем сразу,
    // не заставляя человека набирать то, что он уже поднёс камерой.
    let из_url = "";
    try {
      const p = new URLSearchParams(location.search);
      из_url = p.get("cert") || p.get("check") || "";
    } catch (e) {}
    if (из_url) {
      this.setState({ cert: this.norm(из_url) });
      this.check(this.norm(из_url));
    }
  }

  /** Номер как его вводит человек. Правило одно с приёмником (нормНомер):
   *  пробел вместо дефиса и длинное тире из автозамены не должны ломать поиск. */
  norm(v) {
    return String(v || "").toUpperCase()
      .replace(/[‐-―−]/g, "-")
      .replace(/\s+/g, "-")
      .replace(/-+/g, "-")
      .replace(/^-|-$/g, "");
  }

  onCert(e) {
    this.setState({ cert: e && e.target ? e.target.value : "" });
  }

  onCheck() {
    const n = this.norm(this.state.cert);
    if (!n) return;
    this.setState({ cert: n });
    this.check(n);
  }

  check(номер) {
    this.setState({ busy: true, err: "", res: null });
    fetch(this.PRIEMNIK + "?check=" + encodeURIComponent(номер), { cache: "no-cache" })
      .then(r => r.ok ? r.json() : null)
      .then(d => this.setState({ res: d, busy: false, err: d ? "" : "нет ответа" }))
      // Связи нет — говорим об этом прямо. Молчащая кнопка читается как
      // «сертификат недействителен», а это разные вещи.
      .catch(() => this.setState({ busy: false, err: "нет ответа" }));
  }

  dateRu(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso || ""));
    return m ? (m[3] + "." + m[2] + "." + m[1]) : (iso || "");
  }

  renderVals() {
    const d = this.state.res;
    const rows = [];
    let plash = "", plashBg = "#EEF3F9", plashFg = "#5A6675", hint = "", trusted = false;

    if (this.state.err) {
      plash = "Не удалось проверить";
      hint = "Связь с реестром прервалась. Повторите попытку или напишите нам в WhatsApp +77 000 369 369.";
    } else if (d && d.ok) {
      trusted = !!(d.valid && d.trusted);
      if (d.valid) { plash = "✓ Сертификат действует"; plashBg = "#E8F6EC"; plashFg = "#1B6B34"; }
      else { plash = "Сертификат не действует"; plashBg = "#FDECEA"; plashFg = "#A3251B"; }

      const add = (k, v) => { if (v) rows.push({ k: k, v: String(v) }); };
      add("Кто", d.type === "мастер" ? "Мастер" : "Бригада");
      if (d.type === "мастер") { add("Фамилия и инициалы", d.fio); add("В составе бригады", d.brigada); }
      add("Город", d.city);
      add("Выезд", d.zone);
      add("Направления", (d.dirLabels || []).join(", "));
      add("Квалификация", d.qual);
      add("Электробезопасность", d.eb);
      add("Высота", d.height);
      add("Гарантия бригады на монтаж", d.warranty ? (d.warranty + " мес.") : "");
      add("Работает с", d.since);
      add("Сертификат выдан", this.dateRu(d.issued));
      add("Проверено", d.checked ? (this.dateRu(d.checked) + (d.checkedDays != null ? " · " + d.checkedDays + " дн. назад" : "")) : "");
      add("Действующий бланк", d.issue ? ("выпуск " + d.issue + (d.issueDate ? " от " + this.dateRu(d.issueDate) : "")) : "");
      // Сертификаты производителей приёмник отдаёт только после того, как мы
      // видели документы. Поэтому «предъявлены нам» — факт, а не пересказ.
      if (d.certs && d.certs.list) add("Сертификаты производителей", d.certs.list + " · предъявлены нам " + this.dateRu(d.certs.checked));
      add("Делает", (d.skills || []).map(s => s.title).join(", "));
      if (!d.valid) hint = d.status || "";
    } else if (d && !d.ok) {
      plash = "Номер не найден";
      hint = (d.reason || "") + ". Проверьте номер на бланке. Если он написан верно, напишите нам в WhatsApp +77 000 369 369 — разберёмся.";
    }

    return {
      cert: this.state.cert,
      busy: !!this.state.busy,
      btnText: this.state.busy ? "Проверяем…" : "Проверить",
      hasResult: !!(plash),
      shownCert: (d && d.cert) ? d.cert : this.norm(this.state.cert),
      plash: plash, plashBg: plashBg, plashFg: plashFg,
      rows: rows, hasRows: rows.length > 0,
      trusted: trusted,
      hint: hint, hasHint: !!hint,
      onCert: (e) => this.onCert(e),
      onCheck: () => this.onCheck()
    };
  }
}
