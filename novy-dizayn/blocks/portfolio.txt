    // пустые поля не показываем: карточка с двумя фактами выглядит законченной
    all.forEach(w => {
      w.rows = (w.rows || []).filter(r => r.v && r.v !== need.v);
      // Фото объекта — файл в репозитории (поле photo), а не перетаскивание:

      w.photo = w.photo || "";
      // Галерея: клик по кадру открывает полноэкранный просмотр всех фото объекта
      const g = w.gallery || [];
      w.hasGal = g.length > 0;
      w.galN = g.length;
      w.curs = w.hasGal ? "zoom-in" : "default";
      w.openGal = w.hasGal ? (() => this.setState({ gal: { list: g, i: 0, title: w.title } })) : (() => {});
      w.link = this.linkCard(w.model || "");
      w.hasLink = !!w.link.has;
    });

    const gal = this.state.gal;
    const galStep = d => e => {
      if (e && e.stopPropagation) e.stopPropagation();
      const cur = this.state.gal;
      if (!cur || cur.list.length < 2) return;
      this.setState({ gal: Object.assign({}, cur, { i: (cur.i + d + cur.list.length) % cur.list.length }) });
    };

    // режим правки: владелец открывает страницу с ?edit=1 и видит заготовки и подсказки
    const edit = /(?:^|[?&])edit=1(?:&|$)/.test(location.search);

      galClose: () => this.setState({ gal: null }),
      galPrev: galStep(-1),
      galNext: galStep(1),
      works: all

        .filter(w => f === "Все" || w.kind === f)
    };
