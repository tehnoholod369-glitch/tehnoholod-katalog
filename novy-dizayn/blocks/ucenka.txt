  // Если файл не отдался, пустой белый прямоугольник читается как «фото нет, но
  // всё в порядке». Заменяем его на честную подпись: адрес есть — это ещё не
  // значит, что фото есть (грабли 19.08.2026, 129 мёртвых ссылок из 555).
  componentDidUpdate() {
    document.querySelectorAll("img.th-uc-shot").forEach(img => {
      if (img.__wired) return;
      img.__wired = true;
      img.addEventListener("error", () => {
        const box = img.parentNode;
        img.style.display = "none";
        if (box && !box.querySelector(".th-uc-noshot")) {
          const d = document.createElement("div");
          d.className = "th-uc-noshot";
          d.textContent = "Фото готовится";
          d.style.cssText = "aspect-ratio:4/3; display:flex; align-items:center; justify-content:center; color:#8593A6; font-size:14px; background:#F3F7FB;";
          box.appendChild(d);
        }
      });
    });
  }

  dateRu(iso) {
    const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso || ""));
    return m ? (m[3] + "." + m[2] + "." + m[1]) : (iso || "");
  }

  wa(it) {
    const t = "Здравствуйте! Интересует уценённая позиция № " + it.id
      + " — " + [it.b, it.n].filter(Boolean).join(" ") + ", " + (it.pt || "");
    return "https://wa.me/77000369369?text=" + encodeURIComponent(t);
  }

  plural(n, one, few, many) {
    const a = Math.abs(n) % 100, b = a % 10;
    if (a > 10 && a < 20) return many;
    if (b > 1 && b < 5) return few;
    if (b === 1) return one;
    return many;
  }

  renderVals() {
    const d = this.state.data || {};
    const all = d.items || [];
    const grades = d.grades || [];
    const byCode = {};
    grades.forEach(g => { byCode[g.code] = g; });

    const pickedGrade = this.state.grade;
    const items = pickedGrade ? all.filter(i => i.g === pickedGrade) : all;

    const chips = [];
    if (all.length) {
      chips.push({ code: "", label: "Все", n: all.length });
      grades.forEach(g => {
        const n = all.filter(i => i.g === g.code).length;
        if (n) chips.push({ code: g.code, label: g.code + " · " + g.short, n: n });
      });
    }

    return {
      updated: this.dateRu(d.updated),
      hasItems: all.length > 0,
      isEmpty: this.state.loaded && all.length === 0,

        : "",
      hasChips: chips.length > 2,
      chips: chips.map(c => {
