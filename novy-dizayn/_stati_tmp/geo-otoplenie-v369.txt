  var srcs=["https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/radiatory.json","https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/obogrevateli.json","https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/teplovye-pushki.json","https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/teplovye-zavesy.json","https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/infrakrasnye-obogrevateli.json","https://raw.githubusercontent.com/tehnoholod369-glitch/tehnoholod-katalog/main/novy-dizayn/data/vodyanye-teploventilyatory.json"];
  var bad=false;
  Promise.all(srcs.map(function(u){
    return fetch(u,{cache:"no-cache"}).then(function(r){
      if(!r.ok) throw 0;
      return r.json();
    }).catch(function(){bad=true; return []});
  })).then(function(all){
    if(bad){
      document.querySelectorAll(".thgeo-live").forEach(function(el){
        el.textContent="обновить цены сейчас не удалось — уточните актуальную цену "
          +"по телефону или в WhatsApp";
      });
      return;
    }
    var by={},min=null;
    all.forEach(function(d){(d||[]).forEach(function(x){
      if(x.s)by[x.s]=x;
      if(x.st==="В наличии"&&x.p&&(min===null||x.p<min))min=x.p;
    })});
    document.querySelectorAll(".thgeo-p").forEach(function(td){
      var x=by[td.getAttribute("data-art")];
      if(x&&x.pt)td.textContent=x.pt;
    });
    document.querySelectorAll(".thgeo-pmin").forEach(function(el){
      var a=el.getAttribute("data-min-art"), x=a?by[a]:null, v=(x&&x.p)?x.p:null;
      if(v===null){
        [].forEach.call(document.querySelectorAll(".thgeo-p"),function(td){
          var y=by[td.getAttribute("data-art")];
          if(y&&y.p&&(v===null||y.p<v))v=y.p;
        });
      }
      if(v)el.textContent=v.toLocaleString("ru-RU").replace(/ /g," ");
    });
    document.querySelectorAll(".thgeo-smin").forEach(function(el){
      var x=by[el.getAttribute("data-min-art")];
      if(x&&x.p)el.textContent=x.p.toLocaleString("ru-RU").replace(/ /g," ");
    });
  });
})();
</script>
