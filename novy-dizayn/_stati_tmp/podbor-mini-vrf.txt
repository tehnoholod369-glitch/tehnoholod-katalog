  // расчёт нагрузки помещения: та же формула, что в калькуляторе на /kontakty:
  // объём × 30/35/40 Вт/м³ (+10 % последний этаж, +150/250 Вт большие окна) + люди по 100 Вт + техника
  function load(r) {
    var V = r.a * r.h, q = SUN[r.sun][1];
    var glassW = r.glass ? (r.sun === 2 ? 250 : 150) : 0;
    var env = V * q * (r.top ? 1.1 : 1) + glassW;
    return env + r.ppl * 100 + TECH[r.tech][1];
  }
  // допуск выбора −5 % … +15 % к расчётной нагрузке: берём блок не меньше 95 % нагрузки
  function pickIndoor(r) {
    var list = indoor[r.t] || [];
    if (!list.length) return null;
    var need = load(r) * 0.95;
    for (var i = 0; i < list.length; i++) { if (list[i].kw * 1000 >= need) return { u: list[i], over: false }; }
    return { u: list[list.length - 1], over: true };
  }
